    // VERIFICAÇÃO DO CICLO DE VIDA

    private boolean atividadeDisponivel() {

        return !atividadeEncerrada &&
                !isFinishing() &&
                !isDestroyed();

    // =====================================================
    // VISIBILIDADE DOS CONTROLES
    // =====================================================

    private void atualizarVisibilidadeControles() {

        if (gerenciadorEstadoControles == null) {
            return;
        }

        gerenciadorEstadoControles.atualizar(
                servicoConexao
        );
    }

    // =====================================================
    // MODO IMERSIVO
    // =====================================================

    @Override
    public void onWindowFocusChanged(boolean hasFocus) {

        super.onWindowFocusChanged(hasFocus);

        if (hasFocus && atividadeDisponivel()) {

            ModoImersivo.ativar(getWindow());
        }
    }

    // =====================================================
    // CONEXÃO COMPARTILHADA
    // =====================================================

    private void conectarServidor() {

        servicoConexao = ConexaoActivity.conexao;

        if (servicoConexao == null) {

            status.setText(
                    "ERRO: conexão com o servidor não encontrada."
            );

            Log.e(
                    TAG,
                    "ServicoConexao não foi inicializado."
            );

            atualizarVisibilidadeControles();

            return;
        }

        if (!servicoConexao.estaConectado()) {

            status.setText(
                    "ERRO: conexão com o servidor foi perdida."
            );

            Log.e(
                    TAG,
                    "A conexão compartilhada não está ativa."
            );

            atualizarVisibilidadeControles();

            return;
        }

        gerenciadorComunicacaoServidor.registrarCallbacks(
                servicoConexao
        );

        Log.d(
                TAG,
                "MainActivity reutilizando a conexão existente."
        );

        status.setText(
                "Conectado. Aguardando comandos..."
        );

        atualizarVisibilidadeControles();
    }

    // =====================================================
    // CONTROLE RECEBIDO DO PARCEIRO
    // =====================================================

    private void processarControleRemoto(JSONObject dados) {

        if (!atividadeDisponivel() ||
                gerenciadorControleRemoto == null) {

            return;
        }

        gerenciadorControleRemoto.processar(dados);
    }

    // =====================================================
    // APLICAR CONTROLE REMOTO
    // =====================================================

    private void aplicarControleRemoto(float progressoRecebido) {

        if (!atividadeDisponivel() ||
                movie == null ||
                gerenciadorReproducao == null ||
                gerenciadorControle == null) {

            return;
        }

        controleRemotoAtivo = true;

        pausarAnimacao();

        gerenciadorMovimento.registrarProgresso(
                progressoRecebido,
                gerenciadorControle.isControleManualAtivo(),
                controleRemotoAtivo
        );

        int duracao = obterDuracaoGif();

        gerenciadorReproducao.definirPosicao(
                (long) (progressoRecebido * duracao)
        );

        areaGif.post(() -> {

            if (!atividadeDisponivel() ||
                    movie == null ||
                    gerenciadorControle == null) {

                return;
            }

            gerenciadorControle.moverControles(
                    progressoRecebido
            );

            desenharQuadro();
        });
    }

    // =====================================================
    // LIBERAÇÃO DO CONTROLE REMOTO
    // =====================================================

    private void liberarControleRemoto() {

        controleRemotoAtivo = false;

        Log.d(
                TAG,
                "Parceiro liberou os controles."
        );

        finalizarMovimentoSeLiberado();
    }

    // =====================================================
    // ENVIO DE COMANDOS AO PARCEIRO
    // =====================================================

    private void enviarComandoControle(
            boolean ativo,
            float progresso) {

        if (gerenciadorControleRemoto == null) {
            return;
        }

        gerenciadorControleRemoto.enviar(
                servicoConexao,
                modoIndividual,
                ativo,
                progresso
        );
    }

    // =====================================================
    // DURAÇÃO DO GIF
    // =====================================================

    private int obterDuracaoGif() {

        if (gerenciadorReproducao == null) {
            return 1000;
        }

        return gerenciadorReproducao.obterDuracao(movie);
    }

    // =====================================================
    // DECISÃO APÓS SOLTAR TODOS OS CONTROLES
    // =====================================================

    private void finalizarMovimentoSeLiberado() {

        if (gerenciadorControle == null) {
            return;
        }

        gerenciadorMovimento.finalizarMovimentoSeLiberado(
                gerenciadorControle.isEsquerdoPressionado(),
                gerenciadorControle.isDireitoPressionado(),
                gerenciadorControle.isControleManualAtivo(),
                controleRemotoAtivo
        );
    }

    @Override
    protected void onCreate(Bundle savedInstanceState) {

        super.onCreate(savedInstanceState);

        setContentView(R.layout.activity_main);

        // -------------------------------------------------

                            public void onConectado() {

                                        "Conexão compartilhada ativa."

                            }

                            @Override
                            public void onDesconectado(String motivo) {

                                        TAG,

                                );

                                controleRemotoAtivo = false;

                                finalizarMovimentoSeLiberado();

                            public void onErroConexao(String erro) {

        // -------------------------------------------------
        // MODO DE EXECUÇÃO

        modoIndividual = getIntent().getBooleanExtra(

        // GERENCIADOR DE ESTADO DOS CONTROLES

        gerenciadorEstadoControles =

                        handler

        // -------------------------------------------------

        // -------------------------------------------------

                        "gifs"
                )

        // -------------------------------------------------
        // GERENCIADOR DE CARREGAMENTO DOS GIFS

                                        TAG,

                        new GerenciadorReproducaoGif.Listener() {

                                                !gerenciadorControle.isControleManualAtivo()) &&

                            public void atualizarQuadro() {

                        areaGif,
                        handler,

                            public boolean atividadeDisponivel() {

        // -------------------------------------------------

                                        true,

                                enviarComandoControle(
                                        true,
                                        progresso
                                );
                            }

                            @Override
                            public void onControleManualLiberado() {

                                enviarComandoControle(
                                        false,
                                        0f
                                );

                            }

                );

                        }

        // BOTÃO PRÓXIMO GIF
        // -------------------------------------------------

        botaoProximo.setOnClickListener(v -> {

            gerenciadorMovimento.reiniciarContagem();

            proximoGif();
        });

        configurarControles();

        // -------------------------------------------------
        // CONFIGURAÇÃO DA CONEXÃO
        // -------------------------------------------------

        if (modoIndividual) {

            servicoConexao = null;

            status.setText(
                    "Modo individual. Sem conexão com o servidor."
            );

            Log.d(
                    TAG,
                    "Aplicativo iniciado em modo individual."
            );

        } else {

            conectarServidor();
        }

        atualizarVisibilidadeControles();

        // -------------------------------------------------
        // GERENCIADOR DE CARREGAMENTO DA PLANILHA
        // -------------------------------------------------

        gerenciadorCarregamentoPlanilha =
                new GerenciadorCarregamentoPlanilha(
                        planilhaRepository,
                        executor,
                        handler,
                        new GerenciadorCarregamentoPlanilha.Listener() {

                            @Override
                            public void onCarregamentoConcluido(
                                    GerenciadorCarregamentoPlanilha.Resultado resultado) {

                                if (!atividadeDisponivel()) {
                                    return;
                                }

                                nomes.clear();
                                links.clear();
                                textosPlanilha.clear();

                                nomes.addAll(resultado.getNomes());
                                links.addAll(resultado.getLinks());
                                textosPlanilha.addAll(resultado.getTextos());

                                gerenciadorNavegacaoGif.reiniciar();

                                if (resultado.estaVazio()) {

                                    status.setText(
                                            "Nenhum GIF encontrado."
                                    );

                                    botaoProximo.setEnabled(false);

                                } else {

                                    status.setText(
                                            resultado.getQuantidade() +
                                                    " GIFs disponíveis."
                                    );

                                    botaoProximo.setEnabled(true);

                                    proximoGif();
                                }
                            }

                            @Override
                            public void onErro(Exception erro) {

                                if (!atividadeDisponivel()) {
                                    return;
                                }

                                status.setText(
                                        "Erro ao consultar planilha: " +
                                                erro.getMessage()
                                );

                                botaoProximo.setEnabled(false);

                                Log.e(
                                        TAG,
                                        "Falha no carregamento da planilha.",
                                        erro
                                );
                            }
                        }
                );

        gerenciadorCarregamentoPlanilha.carregar();
    }

    // =====================================================
    // PAUSAR ANIMAÇÃO
    // =====================================================

    private void pausarAnimacao() {

        if (gerenciadorReproducao == null) {
            return;
        }

        gerenciadorReproducao.pausar(movie);
    }

    // =====================================================
    // RETOMAR ANIMAÇÃO
    // =====================================================

    private void iniciarAnimacao() {

        if (gerenciadorReproducao == null) {
            return;
        }

        gerenciadorReproducao.iniciar(movie);
    }

    // =====================================================
    // DESENHO DO QUADRO
    // =====================================================

    private void desenharQuadro() {

        if (!atividadeDisponivel() ||
                movie == null ||
                renderizadorGif == null ||
                gerenciadorReproducao == null) {

            return;
        }

        renderizadorGif.desenhar(
                movie,
                gerenciadorReproducao.getPosicaoMs(),
                pincel
        );
    }

    // =====================================================
    // PRÓXIMO GIF
    // =====================================================

    private void proximoGif() {

        if (!atividadeDisponivel() || carregando) {
            return;
        }

        if (nomes.isEmpty()) {

            status.setText(
                    "Nenhum GIF disponível."
            );

            return;
        }

        int indiceSelecionado =
                gerenciadorNavegacaoGif.proximo(nomes.size());

        if (indiceSelecionado < 0) {
            return;
        }

        baixarEExibir(indiceSelecionado);
    }

    // =====================================================
    // SOLICITAÇÃO DE CARREGAMENTO DO GIF
    // =====================================================

    private void baixarEExibir(int posicao) {

        if (!atividadeDisponivel() || carregando) {
            return;
        }

        if (posicao < 0 ||
                posicao >= nomes.size() ||
                posicao >= links.size()) {

            Log.e(
                    TAG,
                    "Posição inválida para carregamento do GIF: " +
                            posicao
            );

            return;
        }

        nomeGifSolicitado = nomes.get(posicao);

        String link = links.get(posicao);

        status.setText(
                "Carregando: " + nomeGifSolicitado
        );

        gerenciadorCarregamentoGif.carregar(link);
    }

    // =====================================================
    // EXIBIÇÃO DO GIF PRINCIPAL
    // =====================================================

    private void exibirGif(
            File arquivo,
            String nome) {

        if (!atividadeDisponivel()) {
            return;
        }

        if (arquivo == null ||
                !arquivo.isFile() ||
                arquivo.length() == 0) {

            status.setText(
                    "Arquivo GIF inválido."
            );

            return;
        }

        final long geracaoAtual = ++geracaoExibicao;

        // -------------------------------------------------
        // INTERROMPER GIF ANTERIOR
        // -------------------------------------------------

        gerenciadorReproducao.reiniciar();

        gerenciadorTextoPlanilha.parar();

        if (gerenciadorControle != null &&
                gerenciadorControle.isControleManualAtivo()) {

            enviarComandoControle(false, 0f);
        }

        movie = null;

        controleRemotoAtivo = false;

        // -------------------------------------------------
        // REINICIAR CONTROLES E MOVIMENTO
        // -------------------------------------------------

        gerenciadorMovimento.reiniciarContagem();

        gerenciadorControle.limpar();

        atualizarVisibilidadeControles();

        areaGif.post(() -> {

            if (!atividadeDisponivel() ||
                    geracaoAtual != geracaoExibicao) {

                return;
            }

            gerenciadorControle.moverControles(0.5f);
        });

        // -------------------------------------------------
        // DECODIFICAR O GIF
        // -------------------------------------------------

        gerenciadorDecodificacaoGif.decodificar(
                arquivo,
                new GerenciadorDecodificacaoGif.Listener() {

                    @Override
                    public void onDecodificacaoConcluida(
                            Movie resultado) {

                        if (!atividadeDisponivel() ||
                                geracaoAtual != geracaoExibicao) {

                            return;
                        }

                        movie = resultado;

                        gerenciadorReproducao.reiniciar();

                        // ---------------------------------
                        // TEXTO ASSOCIADO AO GIF
                        // ---------------------------------

                        final String textoFinal;

                        int indiceAtual =
                                gerenciadorNavegacaoGif.getIndiceAtual();

                        if (indiceAtual >= 0 &&
                                indiceAtual < textosPlanilha.size()) {

                            textoFinal =
                                    textosPlanilha.get(indiceAtual);

                        } else {

                            textoFinal = "";
                        }

                        final int indiceExibido =
                                indiceAtual + 1;

                        areaGif.post(() -> {

                            if (!atividadeDisponivel() ||
                                    geracaoAtual != geracaoExibicao) {

                                return;
                            }

                            desenharQuadro();

                            iniciarAnimacao();

                            gerenciadorTextoPlanilha.iniciar(
                                    textoFinal
                            );

                            status.setText(
                                    nome + " | " +
                                            indiceExibido +
                                            "/" + nomes.size()
                            );
                        });
                    }

                    @Override
                    public void onErro(Exception erro) {

                        if (!atividadeDisponivel() ||
                                geracaoAtual != geracaoExibicao) {

                            return;
                        }

                        gerenciadorReproducao.reiniciar();

                        gerenciadorTextoPlanilha.parar();

                        movie = null;

                        if (arquivo.exists()) {
                            arquivo.delete();
                        }

                        status.setText(
                                "Erro ao abrir GIF: " +
                                        erro.getMessage()
                        );

                        Log.e(
                                TAG,
                                "Erro ao decodificar GIF.",
                                erro
                        );
                    }
                }
        );
    }

    // =====================================================
    // ENCERRAMENTO
    // =====================================================

    @Override
    protected void onDestroy() {

        atividadeEncerrada = true;

        geracaoExibicao++;

        // -------------------------------------------------
        // ENCERRAR GERENCIADOR DE REPRODUÇÃO
        // -------------------------------------------------

        if (gerenciadorReproducao != null) {

            gerenciadorReproducao.encerrar();
        }

        // -------------------------------------------------
        // ENCERRAR GERENCIADOR DO TEXTO
        // -------------------------------------------------

        if (gerenciadorTextoPlanilha != null) {

            gerenciadorTextoPlanilha.encerrar();
        }

        // -------------------------------------------------
        // REMOVER CALLBACKS PENDENTES
        // -------------------------------------------------

        handler.removeCallbacksAndMessages(null);

        // -------------------------------------------------
        // LIMPAR ESTADO
        // -------------------------------------------------

        movie = null;

        controleRemotoAtivo = false;

        carregando = false;

        nomeGifSolicitado = "";

        gerenciadorMovimento.reiniciarContagem();

        if (gerenciadorControle != null) {

            gerenciadorControle.limpar();
        }

        // -------------------------------------------------
        // REMOVER CALLBACK DA COMUNICAÇÃO
        // -------------------------------------------------

        /*
         * O GerenciadorComunicacaoServidor remove somente
         * o callback registrado pela MainActivity.
         *
         * A conexão WebSocket continua pertencendo
         * à ConexaoActivity.
         */

        if (gerenciadorComunicacaoServidor != null) {

            gerenciadorComunicacaoServidor.removerCallback();
        }

        // -------------------------------------------------
        // LIBERAR RENDERIZADOR
        // -------------------------------------------------

        if (renderizadorGif != null) {

            renderizadorGif.liberar();

            renderizadorGif = null;
        }

        // -------------------------------------------------
        // ENCERRAR EXECUTOR
        // -------------------------------------------------

        executor.shutdownNow();

        super.onDestroy();
    }
}

            );
            return;

        String codigo = campoCodigoPar.getText()

            campoCodigoPar.setError("Digite o código da sala.");
            campoCodigoPar.requestFocus();
            return;

        abriuTelaPrincipal = false;

        botaoConectar.setEnabled(false);
        botaoUsarSemConexao.setEnabled(false);

        statusConexao.setText("Conectando ao servidor...");

        if (conexao == null) {
            conexao = new ServicoConexao(criarCallback());
        } else {
            conexao.definirCallback(criarCallback());
        }

        conexao.conectar(codigo);
    }

    private ServicoConexao.Callback criarCallback() {

        return new ServicoConexao.Callback() {

            @Override
            public void onConectado() {
                runOnUiThread(() -> {
                    if (isFinishing() || isDestroyed()) return;

                    Log.d(TAG, "Conexão WebSocket estabelecida.");

                    statusConexao.setText(
                            "Servidor conectado. Entrando na sala..."
                    );
                });
            }

            @Override
            public void onDesconectado(String motivo) {
                runOnUiThread(() -> {
                    if (isFinishing() || isDestroyed()) return;

                    if (!abriuTelaPrincipal) {
                        statusConexao.setText(
                                "Conexão encerrada. Tente novamente."
                        );
                        liberarTelaConexao();
                    }
                });
            }

            @Override
            public void onMensagem(JSONObject mensagem) {
                runOnUiThread(() -> {
                    if (isFinishing() || isDestroyed()) return;
                    processarMensagem(mensagem);
                });
            }

            @Override
            public void onErro(String erro) {
                runOnUiThread(() -> {
                    if (isFinishing() || isDestroyed()) return;

                    Log.e(TAG, erro);

                    if (!abriuTelaPrincipal) {
                        statusConexao.setText("Erro: " + erro);
                        liberarTelaConexao();
                    }
                });
            }
        };
    }

    private void processarMensagem(JSONObject mensagem) {

        String tipo = mensagem.optString("type", "");

        Log.d(TAG, "Evento recebido do servidor: " + mensagem);

        switch (tipo) {

            case "joined": {
                int usuarios = mensagem.optInt("users", 0);

                if (usuarios >= 2) {
                    abrirTelaPrincipal();
                } else {
                    statusConexao.setText(
                            "Conectado! Aguardando o segundo usuário..."
                    );
                }
                break;
            }

            case "peer_connected":
                statusConexao.setText(
                        "Parceiro conectado! Abrindo aplicativo..."
                );
                abrirTelaPrincipal();
                break;

            case "peer_disconnected":
                statusConexao.setText("Seu parceiro desconectou.");
                break;

            case "error": {
                String erro = mensagem.optString(
                        "message",
                        "O servidor informou um erro."
                );
                statusConexao.setText(erro);
                liberarTelaConexao();
                break;
            }

            default:
                Log.d(TAG, "Evento não tratado: " + tipo);
                break;
        }
    }

    private void abrirTelaPrincipal() {

        if (abriuTelaPrincipal) return;

        if (conexao == null || !conexao.estaConectado()) {
            statusConexao.setText(
                    "A conexão com o servidor foi perdida."
            );
            liberarTelaConexao();
            return;
        }

        abriuTelaPrincipal = true;

        Intent intent = new Intent(
                ConexaoActivity.this,
                MainActivity.class
        );

        intent.putExtra("codigo_par", conexao.getCodigoPar());
        intent.putExtra("papel_usuario", conexao.getPapelUsuario());
        startActivity(intent);
    }

    private void liberarTelaConexao() {

        botaoConectar.setEnabled(true);
        botaoUsarSemConexao.setEnabled(true);
        campoCodigoPar.setEnabled(true);
    }

    @Override
    protected void onResume() {

        super.onResume();
        ativarModoImersivo();

        if (!abriuTelaPrincipal && conexao != null) {
            conexao.definirCallback(criarCallback());
        }
    }

    @Override
    protected void onDestroy() {
        super.onDestroy();
        // A conexão permanece compartilhada com MainActivity.
    }
}

package com.example.rpghqconspiration;

import android.os.Handler;
import android.os.Looper;
import android.util.Log;

import org.json.JSONException;
import org.json.JSONObject;

import java.net.URI;
import java.util.concurrent.atomic.AtomicBoolean;

import org.java_websocket.client.WebSocketClient;
import org.java_websocket.enums.ReadyState;
import org.java_websocket.handshake.ServerHandshake;

public class ServicoConexao {

    private static final String TAG = "ServicoConexao";

    private static final String URL_SERVIDOR =
            "wss://rpghqserver-1.onrender.com";

    public interface Callback {
        void onConectado();
        void onDesconectado(String motivo);
        void onMensagem(JSONObject mensagem);
        void onErro(String erro);
    }

    private final Handler handlerPrincipal =
            new Handler(Looper.getMainLooper());

    private volatile Callback callback;
    private volatile WebSocketClient cliente;

    private volatile boolean conectado = false;
    private volatile String codigoPar;
    private volatile String papelUsuario = "";

    private volatile boolean controleConcedido = false;
    private volatile boolean solicitacaoPendente = false;

    private final AtomicBoolean entradaEnviada =
            new AtomicBoolean(false);

    public ServicoConexao(Callback callback) {
        this.callback = callback;
    }

    public void definirCallback(Callback callback) {
        this.callback = callback;
    }

    // =====================================================
    // CONEXÃO
    // =====================================================

    public synchronized void conectar(String codigo) {

        if (codigo == null || codigo.trim().isEmpty()) {
            avisarErro("Digite o código da sala.");
            return;
        }

        if (estaConectado()) {
            Log.d(TAG, "Já existe uma conexão ativa.");
            avisarConectado();
            return;
        }

        if (cliente != null) {
            ReadyState estado = cliente.getReadyState();

            if (estado == ReadyState.OPEN
                    || estado == ReadyState.NOT_YET_CONNECTED) {
                Log.d(TAG, "Conexão já está em andamento.");
                return;
            }
        }

        codigoPar = codigo.trim();
        papelUsuario = "";

        conectado = false;
        controleConcedido = false;
        solicitacaoPendente = false;

        entradaEnviada.set(false);

        try {

            cliente = new WebSocketClient(new URI(URL_SERVIDOR)) {

                @Override
                public void onOpen(ServerHandshake handshake) {

                    conectado = true;

                    Log.d(TAG, "WebSocket conectado.");

                    enviarEntradaNaSala();
                    avisarConectado();
                }

                @Override
                public void onMessage(String mensagemRecebida) {

                    Log.d(
                            TAG,
                            "Mensagem recebida: " + mensagemRecebida
                    );

                    try {

                        JSONObject mensagem =
                                new JSONObject(mensagemRecebida);

                        processarEstadoDaConexao(mensagem);
                        avisarMensagem(mensagem);

                    } catch (JSONException e) {

                        Log.e(
                                TAG,
                                "Mensagem JSON inválida.",
                                e
                        );

                        avisarErro("Mensagem recebida inválida.");
                    }
                }

                @Override
                public void onClose(
                        int codigoFechamento,
                        String motivo,
                        boolean remoto) {

                    conectado = false;
                    controleConcedido = false;
                    solicitacaoPendente = false;

                    entradaEnviada.set(false);

                    Log.d(
                            TAG,
                            "WebSocket encerrado: " + motivo
                    );

                    avisarDesconectado(motivo);
                }

                @Override
                public void onError(Exception erro) {

                    conectado = false;
                    controleConcedido = false;
                    solicitacaoPendente = false;

                    Log.e(TAG, "Erro no WebSocket.", erro);

                    String mensagem = erro.getMessage();

                    if (mensagem == null
                            || mensagem.trim().isEmpty()) {

                        mensagem =
                                "Erro na conexão com o servidor.";
                    }

                    avisarErro(mensagem);
                }
            };

            cliente.connect();

        } catch (Exception e) {

            conectado = false;
            controleConcedido = false;
            solicitacaoPendente = false;

            Log.e(
                    TAG,
                    "Não foi possível iniciar a conexão.",
                    e
            );

            avisarErro(
                    "Não foi possível iniciar a conexão: "
                            + e.getMessage()
            );
        }
    }

    // =====================================================
    // PROCESSAMENTO DAS MENSAGENS DO SERVIDOR
    // =====================================================

    private void processarEstadoDaConexao(JSONObject mensagem) {

        String tipo = mensagem.optString("type", "");

        switch (tipo) {

            case "joined":

                papelUsuario =
                        mensagem.optString("role", "");

                Log.d(
                        TAG,
                        "Papel atribuído pelo servidor: "
                                + papelUsuario
                );

                break;

            case "control_granted":

                controleConcedido = true;
                solicitacaoPendente = false;

                Log.d(
                        TAG,
                        "Controle autorizado pelo servidor."
                );

                break;

            case "control_busy":

                controleConcedido = false;
                solicitacaoPendente = false;

                Log.d(
                        TAG,
                        "Controle ocupado pelo outro usuário."
                );

                break;

            case "control_released":

                controleConcedido = false;
                solicitacaoPendente = false;

                Log.d(
                        TAG,
                        "Controle liberado pelo servidor."
                );

                break;

            case "peer_disconnected":

                controleConcedido = false;
                solicitacaoPendente = false;

                Log.d(
                        TAG,
                        "O outro usuário desconectou."
                );

                break;

            case "control":

                if (!mensagem.optBoolean("active", false)) {

                    controleConcedido = false;
                    solicitacaoPendente = false;
                }

                break;

            case "error":

                solicitacaoPendente = false;

                Log.e(
                        TAG,
                        "Erro informado pelo servidor: "
                                + mensagem.optString("message", "")
                );

                break;
        }
    }

    // =====================================================
    // ENTRADA NA SALA
    // =====================================================

    private void enviarEntradaNaSala() {

        if (!conectado
                || codigoPar == null
                || !entradaEnviada.compareAndSet(false, true)) {

            return;
        }

        try {

            JSONObject mensagem = new JSONObject();

            mensagem.put("type", "join");
            mensagem.put("pair", codigoPar);

            enviarJson(mensagem);

            Log.d(
                    TAG,
                    "Entrada na sala enviada: " + codigoPar
            );

        } catch (JSONException e) {

            entradaEnviada.set(false);

            Log.e(
                    TAG,
                    "Erro ao preparar entrada na sala.",
                    e
            );

            avisarErro("Erro ao entrar na sala.");
        }
    }

    // =====================================================
    // ENVIAR CONTROLE
    // =====================================================

    public void enviarControle(boolean ativo, float progresso) {

        if (!estaConectado()) {

            Log.w(
                    TAG,
                    "Controle não enviado: conexão indisponível."
            );

            return;
        }

        if (ativo && !controleConcedido) {
            solicitacaoPendente = true;
        }

        if (!ativo) {
            controleConcedido = false;
            solicitacaoPendente = false;
        }

        float progressoLimitado = Math.max(
                0.0f,
                Math.min(1.0f, progresso)
        );

        try {

            JSONObject mensagem = new JSONObject();

            mensagem.put("type", "control");
            mensagem.put("active", ativo);
            mensagem.put("progress", progressoLimitado);

            enviarJson(mensagem);

            Log.d(
                    TAG,
                    "Controle enviado: " + mensagem
            );

        } catch (JSONException e) {

            Log.e(
                    TAG,
                    "Erro ao preparar comando de controle.",
                    e
            );

            avisarErro(
                    "Não foi possível preparar o comando."
            );
        }
    }

    // =====================================================
    // ENVIO JSON
    // =====================================================

    private synchronized void enviarJson(JSONObject mensagem) {

        WebSocketClient conexaoAtual = cliente;

        if (conexaoAtual == null
                || conexaoAtual.getReadyState()
                != ReadyState.OPEN) {

            Log.w(
                    TAG,
                    "Mensagem não enviada: WebSocket fechado."
            );

            return;
        }

        try {

            conexaoAtual.send(mensagem.toString());

            Log.d(
                    TAG,
                    "Mensagem enviada: " + mensagem
            );

        } catch (Exception e) {

            Log.e(
                    TAG,
                    "Erro ao enviar mensagem.",
                    e
            );

            avisarErro(
                    "Falha ao enviar dados ao parceiro."
            );
        }
    }

    // =====================================================
    // CONSULTAS DE ESTADO
    // =====================================================

    public boolean estaConectado() {

        WebSocketClient conexaoAtual = cliente;

        return conectado
                && conexaoAtual != null
                && conexaoAtual.getReadyState()
                == ReadyState.OPEN;
    }

    public String getCodigoPar() {
        return codigoPar;
    }

    public String getPapelUsuario() {
        return papelUsuario;
    }

    public boolean possuiControle() {
        return controleConcedido;
    }

    public boolean possuiSolicitacaoPendente() {
        return solicitacaoPendente;
    }

    // =====================================================
    // DESCONECTAR
    // =====================================================

    public synchronized void desconectar() {

        conectado = false;
        controleConcedido = false;
        solicitacaoPendente = false;

        entradaEnviada.set(false);

        WebSocketClient conexaoAtual = cliente;
        cliente = null;

        if (conexaoAtual != null) {

            try {

                if (conexaoAtual.isOpen()) {
                    conexaoAtual.close();
                }

            } catch (Exception e) {

                Log.e(
                        TAG,
                        "Erro ao fechar WebSocket.",
                        e
                );
            }
        }
    }

    // =====================================================
    // CALLBACKS
    // =====================================================

    private void avisarConectado() {

        handlerPrincipal.post(() -> {

            Callback atual = callback;

            if (atual != null) {
                atual.onConectado();
            }
        });
    }

    private void avisarDesconectado(String motivo) {

        handlerPrincipal.post(() -> {

            Callback atual = callback;

            if (atual != null) {
                atual.onDesconectado(motivo);
            }
        });
    }

    private void avisarMensagem(JSONObject mensagem) {

        handlerPrincipal.post(() -> {

            Callback atual = callback;

            if (atual != null) {
                atual.onMensagem(mensagem);
            }
        });
    }

    private void avisarErro(String erro) {

        handlerPrincipal.post(() -> {

            Callback atual = callback;

            if (atual != null) {
                atual.onErro(erro);
            }
        });
    }

package com.example.rpghqconspiration;

import android.content.Context;
import android.graphics.Canvas;
import android.graphics.Color;
import android.graphics.Movie;
import android.graphics.Paint;
import android.util.AttributeSet;

import androidx.appcompat.widget.AppCompatImageView;

import java.io.File;
import java.io.FileInputStream;

public class GifSincronizadoView extends AppCompatImageView {

    private Movie movie;

    private final Paint paint = new Paint(
            Paint.ANTI_ALIAS_FLAG |
                    Paint.FILTER_BITMAP_FLAG
    );

    private int duracaoMs = 1000;
    private int posicaoMs = 0;
    private boolean reproduzindo = false;

    public GifSincronizadoView(Context context) {
        super(context);
        inicializar();
    }

    public GifSincronizadoView(
            Context context,
            AttributeSet attrs
    ) {
        super(context, attrs);
        inicializar();
    }

    public GifSincronizadoView(
            Context context,
            AttributeSet attrs,
            int defStyleAttr
    ) {
        super(context, attrs, defStyleAttr);
        inicializar();
    }

    private void inicializar() {
        setLayerType(LAYER_TYPE_SOFTWARE, null);
        setBackgroundColor(Color.BLACK);
        setScaleType(ScaleType.FIT_CENTER);
    }

    public synchronized boolean carregarGif(File arquivo) {

        if (arquivo == null ||
                !arquivo.exists() ||
                arquivo.length() <= 0) {
            return false;
        }

        try (FileInputStream entrada =
                     new FileInputStream(arquivo)) {

            Movie novoMovie = Movie.decodeStream(entrada);

            if (novoMovie == null) {
                return false;
            }

            movie = novoMovie;

            int duracao = movie.duration();

            duracaoMs = duracao > 0 ? duracao : 1000;

            posicaoMs = 0;
            reproduzindo = false;

            postInvalidate();

            return true;

        } catch (Exception e) {

            movie = null;

            return false;
        }
    }

    public synchronized void definirPosicao(int posicao) {

        if (movie == null) return;

        posicaoMs = limitarPosicao(posicao);

        postInvalidate();
    }

    public synchronized void iniciar() {

        if (movie == null) return;

        reproduzindo = true;

        postInvalidate();
    }

    public synchronized void pausar() {

        reproduzindo = false;

        postInvalidate();
    }

    public synchronized void reiniciar() {

        posicaoMs = 0;

        reproduzindo = false;

        postInvalidate();
    }

    public synchronized int getDuracaoMs() {
        return duracaoMs;
    }

    public synchronized int getPosicaoMs() {
        return posicaoMs;
    }

    public synchronized boolean estaReproduzindo() {
        return reproduzindo;
    }

    public synchronized boolean possuiGif() {
        return movie != null;
    }

    private int limitarPosicao(int posicao) {

        return Math.max(
                0,
                Math.min(posicao, duracaoMs - 1)
        );
    }

    @Override
    protected synchronized void onDraw(Canvas canvas) {

        if (movie == null) {
            super.onDraw(canvas);
            return;
        }

        canvas.drawColor(Color.BLACK);

        int largura = getWidth();
        int altura = getHeight();

        if (largura <= 0 || altura <= 0) return;

        int larguraGif = movie.width();
        int alturaGif = movie.height();

        if (larguraGif <= 0 || alturaGif <= 0) return;

        movie.setTime(limitarPosicao(posicaoMs));

        float escala = Math.min(
                (float) largura / larguraGif,
                (float) altura / alturaGif
        );

        float x = (largura - larguraGif * escala) / 2f;
        float y = (altura - alturaGif * escala) / 2f;

        canvas.save();
        canvas.translate(x, y);
        canvas.scale(escala, escala);

        movie.draw(canvas, 0, 0, paint);

        canvas.restore();
    }
}

    // INTERFACE DE COMUNICAÇÃO COM A MAINACTIVITY
    // =====================================================

    public interface Listener {

        boolean atividadeDisponivel();

        void onConectado();

        void onDesconectado(String motivo);

        void onMensagem(JSONObject mensagem);

        void onErro(String erro);
    }

    // CONSTRUTOR
    // =====================================================

    public GerenciadorSessao() {
    }

    // =====================================================
    // REGISTRAR CALLBACKS
    // =====================================================

    public void registrarCallbacks(
            ServicoConexao servico,
            Listener novoListener) {

        if (servico == null || novoListener == null) {

            Log.e(TAG, "Serviço ou listener inválido.");

            return;
        }

        this.servicoConexao = servico;
        this.listener = novoListener;

        callbackRegistrado = new ServicoConexao.Callback() {

            @Override
            public void onConectado() {

                Listener atual = obterListenerAtivo();

                if (atual != null) {
                    atual.onConectado();
                }
            }

            @Override
            public void onDesconectado(String motivo) {

                Listener atual = obterListenerAtivo();

                if (atual != null) {
                    atual.onDesconectado(motivo);
                }
            }

            @Override
            public void onMensagem(JSONObject mensagem) {

                Listener atual = obterListenerAtivo();

                if (atual != null) {
                    atual.onMensagem(mensagem);
                }
            }

            @Override
            public void onErro(String erro) {

                Listener atual = obterListenerAtivo();

                if (atual != null) {
                    atual.onErro(erro);
                }
            }
        };

        servicoConexao.definirCallback(callbackRegistrado);

        Log.d(TAG, "Callbacks registrados.");
    }

    // =====================================================
    // VERIFICAR DISPONIBILIDADE DA ATIVIDADE
    // =====================================================

    private Listener obterListenerAtivo() {

        Listener atual = listener;

        if (atual == null) {
            return null;
        }

        if (!atual.atividadeDisponivel()) {
            return null;
        }

        return atual;
    }

    // =====================================================
    // REMOVER CALLBACK
    // =====================================================

    public void removerCallback() {

        if (servicoConexao != null) {

            servicoConexao.definirCallback(null);
        }

        callbackRegistrado = null;

        listener = null;

        servicoConexao = null;

        Log.d(TAG, "Callbacks removidos.");
    }
}

package com.example.rpghqconspiration;

import android.util.Log;

/**
 * Gerencia a contagem dos ciclos de movimentação
 * e decide o que fazer quando os controles são liberados.
 *
 * Não controla diretamente Views, MotionEvents ou GIFs.
 */
public final class GerenciadorMovimento {

    private static final String TAG = "GerenciadorMovimento";

    // =====================================================
    // CONFIGURAÇÕES
    // =====================================================

    private static final int CICLOS_NECESSARIOS = 3;

    private static final float LIMITE_EXTREMO_SUPERIOR = 0.15f;

    private static final float LIMITE_EXTREMO_INFERIOR = 0.85f;

    // =====================================================
    // ESTADO DA MOVIMENTAÇÃO
    // =====================================================

    private int ciclosConcluidos = 0;

    private int extremoInicial = -1;

    private boolean alcancouExtremoOposto = false;

    // =====================================================
    // COMUNICAÇÃO COM A MAINACTIVITY
    // =====================================================

    public interface Listener {

        boolean possuiGif();

        void avancarParaProximoGif();

        void retomarAnimacao();
    }

    private final Listener listener;

    // =====================================================
    // CONSTRUTOR
    // =====================================================

    public GerenciadorMovimento(Listener listener) {

        if (listener == null) {
            throw new IllegalArgumentException(
                    "O Listener não pode ser nulo."
            );
        }

        this.listener = listener;
    }

    // =====================================================
    // REGISTRAR PROGRESSO DO MOVIMENTO
    // =====================================================

    public void registrarProgresso(
            float progresso,
            boolean controleManualAtivo,
            boolean controleRemotoAtivo) {

        if (!controleManualAtivo &&
                !controleRemotoAtivo) {

            return;
        }

        if (Float.isNaN(progresso) ||
                Float.isInfinite(progresso)) {

            return;
        }

        progresso = Math.max(
                0f,
                Math.min(1f, progresso)
        );

        int extremoAtual = -1;

        if (progresso <= LIMITE_EXTREMO_SUPERIOR) {

            extremoAtual = 0;

        } else if (progresso >= LIMITE_EXTREMO_INFERIOR) {

            extremoAtual = 1;
        }

        if (extremoAtual == -1) {
            return;
        }

        if (extremoInicial == -1) {

            extremoInicial = extremoAtual;

            return;
        }

        if (!alcancouExtremoOposto &&
                extremoAtual != extremoInicial) {

            alcancouExtremoOposto = true;

            return;
        }

        if (alcancouExtremoOposto &&
                extremoAtual == extremoInicial) {

            ciclosConcluidos++;

            alcancouExtremoOposto = false;

            Log.d(
                    TAG,
                    "Ciclo completo: " +
                            ciclosConcluidos +
                            "/" +
                            CICLOS_NECESSARIOS
            );
        }
    }

    // =====================================================
    // REINICIAR CONTAGEM
    // =====================================================

    public void reiniciarContagem() {

        ciclosConcluidos = 0;

        extremoInicial = -1;

        alcancouExtremoOposto = false;

        Log.d(
                TAG,
                "Contagem de movimentos reiniciada."
        );
    }

    // =====================================================
    // FINALIZAR MOVIMENTO
    // =====================================================

    public void finalizarMovimentoSeLiberado(
            boolean esquerdoPressionado,
            boolean direitoPressionado,
            boolean controleManualAtivo,
            boolean controleRemotoAtivo) {

        // Ainda existe algum controle em utilização.

        if (esquerdoPressionado ||
                direitoPressionado ||
                controleManualAtivo ||
                controleRemotoAtivo) {

            return;
        }

        // Não existe GIF carregado.

        if (!listener.possuiGif()) {

            reiniciarContagem();

            return;
        }

        // Movimento suficiente para avançar.

        if (ciclosConcluidos >= CICLOS_NECESSARIOS) {

            Log.d(
                    TAG,
                    "Três ciclos concluídos. Avançando para o próximo GIF."
            );

            reiniciarContagem();

            listener.avancarParaProximoGif();

            return;
        }

        // Movimento insuficiente: mantém o GIF atual.

        Log.d(
                TAG,
                "Movimento insuficiente: " +
                        ciclosConcluidos +
                        "/" +
                        CICLOS_NECESSARIOS +
                        ". Mantendo o GIF atual."
        );

        reiniciarContagem();

        listener.retomarAnimacao();
    }

    // =====================================================
    // CONSULTA DO PROGRESSO
    // =====================================================

    public int getCiclosConcluidos() {

        return ciclosConcluidos;
    }

    public int getCiclosNecessarios() {

        return CICLOS_NECESSARIOS;
    }
}

package com.example.rpghqconspiration;

import java.util.ArrayList;

/**
 * Responsável por interpretar linhas no formato CSV.
 *
 * Identifica vírgulas que separam colunas, respeitando
 * vírgulas dentro de campos delimitados por aspas.
 */
public final class CsvParser {

    private CsvParser() {
        // Impede a criação de instâncias.
    }

    /**
     * Separa uma linha CSV em suas respectivas colunas.
     *
     * Também interpreta aspas duplas escapadas ("").
     */
    public static ArrayList<String> separar(String linha) {

        ArrayList<String> resultado =
                new ArrayList<>();

        StringBuilder campo =
                new StringBuilder();

        boolean dentroAspas = false;

        for (int i = 0; i < linha.length(); i++) {

            char caractere = linha.charAt(i);

            if (caractere == '"') {

                if (dentroAspas &&
                        i + 1 < linha.length() &&
                        linha.charAt(i + 1) == '"') {

                    campo.append('"');

                    i++;

                } else {

                    dentroAspas = !dentroAspas;
                }

            } else if (caractere == ',' &&
                    !dentroAspas) {

                resultado.add(campo.toString());

                campo.setLength(0);

            } else {

                campo.append(caractere);
            }
        }

        resultado.add(campo.toString());

        return resultado;
    }
}

            String resultado = null;

                return null;
            }

            resultado = resultado.trim();

            if (resultado.isEmpty() ||
                    !resultado.matches("[a-zA-Z0-9_-]+")) {

                return null;
            }

            return resultado;

        } catch (Exception e) {

            return null;
        }
    }

/**
 * Responsável por baixar e validar arquivos GIF
 * armazenados no Google Drive.
 *
 * Não manipula Views nem controla a navegação.
 */
public final class GoogleDriveDownloader {

    private GoogleDriveDownloader() {
        // Impede a criação de instâncias.
    }

    /**
     * Baixa um GIF e retorna o arquivo salvo.
     *
     * @param id ID do arquivo no Google Drive.
     * @param pastaGifs Pasta onde os GIFs são armazenados.
     */
    public static File baixar(
            String id,
            File pastaGifs) throws Exception {

        if (id == null || id.trim().isEmpty()) {
            throw new Exception("ID do arquivo inválido.");
        }

        if (pastaGifs == null) {
            throw new Exception("Pasta de GIFs inválida.");
        }

        if (!pastaGifs.exists() && !pastaGifs.mkdirs()) {
            throw new Exception("Não foi possível criar a pasta de GIFs.");
        }

        File arquivo = new File(
                pastaGifs,
                id + ".gif"
        );

        File temporario = new File(
                pastaGifs,
                id + ".part"
        );

        HttpURLConnection conexao = null;

        try {

            String endereco =
                    "https://drive.usercontent.google.com/download?id=" +
                            id +
                            "&export=download&confirm=t";

            conexao = (HttpURLConnection)
                    new URL(endereco).openConnection();

            conexao.setConnectTimeout(20000);
            conexao.setReadTimeout(30000);
            conexao.setInstanceFollowRedirects(true);

            int codigoHttp = conexao.getResponseCode();

            if (codigoHttp != HttpURLConnection.HTTP_OK) {
                throw new Exception("Erro HTTP: " + codigoHttp);
            }

            try (
                    InputStream entrada = conexao.getInputStream();

                    FileOutputStream saida =
                            new FileOutputStream(temporario)
            ) {

                byte[] buffer = new byte[8192];

                int quantidade;

                while ((quantidade = entrada.read(buffer)) != -1) {
                    saida.write(buffer, 0, quantidade);
                }

                saida.flush();
            }

            if (temporario.length() == 0) {
                throw new Exception("Arquivo vazio.");
            }

            try (
                    FileInputStream entradaValidacao =
                            new FileInputStream(temporario)
            ) {

                Movie teste = Movie.decodeStream(
                        entradaValidacao
                );

                if (teste == null) {
                    throw new Exception(
                            "O arquivo recebido não é um GIF válido."
                    );
                }
            }

            if (arquivo.exists() && !arquivo.delete()) {
                throw new Exception(
                        "Não foi possível substituir o arquivo antigo."
                );
            }

            if (!temporario.renameTo(arquivo)) {
                throw new Exception("Erro ao salvar GIF.");
            }

            return arquivo;

        } catch (Exception e) {

            if (temporario.exists()) {
                temporario.delete();
            }

            throw e;

        } finally {

            if (conexao != null) {
                conexao.disconnect();
            }
        }
    }
}

package com.example.rpghqconspiration;

import android.graphics.Bitmap;
import android.graphics.Canvas;
import android.graphics.Color;
import android.graphics.Movie;
import android.graphics.Paint;
import android.widget.ImageView;

/**
 * Responsável por desenhar os quadros dos GIFs.
 *
 * Controla o Bitmap e o Canvas utilizados
 * para apresentar a animação na ImageView.
 */
public class GifFrameRenderer {

    private final ImageView imagemGif;

    private Bitmap quadroBitmap;
    private Canvas quadroCanvas;

    public GifFrameRenderer(ImageView imagemGif) {
        this.imagemGif = imagemGif;
    }

    public void desenhar(
            Movie movie,
            long posicaoMs,
            Paint pincel) {

        if (movie == null ||
                imagemGif == null ||
                imagemGif.getWidth() <= 0 ||
                imagemGif.getHeight() <= 0) {

            return;
        }

        int largura = imagemGif.getWidth();
        int altura = imagemGif.getHeight();

        if (quadroBitmap == null ||
                quadroBitmap.getWidth() != largura ||
                quadroBitmap.getHeight() != altura) {

            liberarBitmap();

            quadroBitmap = Bitmap.createBitmap(
                    largura,
                    altura,
                    Bitmap.Config.ARGB_8888
            );

            quadroCanvas = new Canvas(quadroBitmap);

            imagemGif.setImageBitmap(quadroBitmap);
        }

        quadroCanvas.drawColor(Color.BLACK);

        int larguraGif = movie.width();
        int alturaGif = movie.height();

        if (larguraGif <= 0 || alturaGif <= 0) {
            return;
        }

        float escala = Math.min(
                (float) largura / larguraGif,
                (float) altura / alturaGif
        );

        float larguraFinal = larguraGif * escala;
        float alturaFinal = alturaGif * escala;

        float deslocamentoX =
                (largura - larguraFinal) / 2f;

        float deslocamentoY =
                (altura - alturaFinal) / 2f;

        quadroCanvas.save();

        quadroCanvas.translate(
                deslocamentoX,
                deslocamentoY
        );

        quadroCanvas.scale(escala, escala);

        movie.setTime((int) posicaoMs);

        movie.draw(
                quadroCanvas,
                0,
                0,
                pincel
        );

        quadroCanvas.restore();

        imagemGif.invalidate();
    }

    public void liberar() {

        liberarBitmap();

        quadroCanvas = null;
    }

    private void liberarBitmap() {

        if (quadroBitmap != null) {

            if (!quadroBitmap.isRecycled()) {
                quadroBitmap.recycle();
            }

            quadroBitmap = null;
        }
    }
}

public final class GerenciadorControleMultitoque {

    private static final String TAG =
            "GerenciadorMultitoque";

    // =====================================================
    // COMUNICAÇÃO COM A MAINACTIVITY
    // =====================================================

    public interface Listener {

        boolean possuiGif();

        void onControleManualIniciado();

        void onProgressoManual(float progresso);

    private final Listener listener;

    // ESTADO DOS DEDOS
    // =====================================================

    private boolean direitoPressionado = false;

    private boolean controleManualAtivo = false;

    private float posicaoDedoEsquerdo = 0;

    private float posicaoDedoDireito = 0;

    private int idDedoEsquerdo = -1;

    private int idDedoDireito = -1;

    // =====================================================
    // CONSTRUTOR
    // =====================================================

    public GerenciadorControleMultitoque(
            FrameLayout areaGif,
            TextView controleEsquerdo,
            TextView controleDireito,
            Listener listener) {

        if (areaGif == null ||
                controleEsquerdo == null ||
                controleDireito == null ||
                listener == null) {

            throw new IllegalArgumentException(
                    "Os componentes e o Listener são obrigatórios."
            );
        }

        this.areaGif = areaGif;
        this.controleEsquerdo = controleEsquerdo;
        this.controleDireito = controleDireito;
        this.listener = listener;
    }

    // =====================================================
    // CONFIGURAÇÃO DOS CONTROLES
    // =====================================================

    public void configurar() {

        areaGif.setMotionEventSplittingEnabled(false);

        View.OnTouchListener listenerMultitoque =
                (view, evento) -> {

                    processarEventoMultitoque(
                            view,
                            evento
                    );

                    return true;
                };

        controleEsquerdo.setOnTouchListener(
                listenerMultitoque
        );

        controleDireito.setOnTouchListener(
                listenerMultitoque
        );
    }

    // =====================================================
    // PROCESSAMENTO MULTITOQUE
    // =====================================================

    private void processarEventoMultitoque(
            View view,
            MotionEvent evento) {

        int acao = evento.getActionMasked();

        int indice = evento.getActionIndex();

        switch (acao) {

            case MotionEvent.ACTION_DOWN:

            case MotionEvent.ACTION_POINTER_DOWN:

                atribuirPonteiro(
                        view,
                        evento,
                        indice
                );

                verificarControleManual();

                atualizarControleManual();

                break;

            case MotionEvent.ACTION_MOVE:

                atualizarPosicoesDedos(evento);

                atualizarControleManual();

                break;

            case MotionEvent.ACTION_POINTER_UP:

                atualizarPosicoesDedos(evento);

                int id = evento.getPointerId(indice);

                liberarPonteiro(id);

                verificarLiberacao();

                break;

            case MotionEvent.ACTION_UP:

            case MotionEvent.ACTION_CANCEL:

                liberarTodosOsDedos();

                verificarLiberacao();

                break;
        }
    }

    // =====================================================
    // COORDENADAS NA TELA
    // =====================================================

    private float obterXNaTela(
            MotionEvent evento,
            int indice) {

        float deslocamentoX =
                evento.getRawX() - evento.getX(0);

        return evento.getX(indice) + deslocamentoX;
    }

    private float obterYNaTela(
            MotionEvent evento,
            int indice) {

        float deslocamentoY =
                evento.getRawY() - evento.getY(0);

        return evento.getY(indice) + deslocamentoY;
    }

    // =====================================================
    // IDENTIFICAÇÃO DO CONTROLE
    // =====================================================

    private int identificarControle(
            float x,
            float y) {

        int[] localizacao = new int[2];

        controleEsquerdo.getLocationOnScreen(
                localizacao
        );

        if (x >= localizacao[0] &&
                x <= localizacao[0] +
                        controleEsquerdo.getWidth() &&
                y >= localizacao[1] &&
                y <= localizacao[1] +
                        controleEsquerdo.getHeight()) {

            return 1;
        }

        controleDireito.getLocationOnScreen(
                localizacao
        );

        if (x >= localizacao[0] &&
                x <= localizacao[0] +
                        controleDireito.getWidth() &&
                y >= localizacao[1] &&
                y <= localizacao[1] +
                        controleDireito.getHeight()) {

            return 2;
        }

        return 0;
    }

    // =====================================================
    // ATRIBUI UM DEDO A UM CONTROLE
    // =====================================================

    private void atribuirPonteiro(
            View view,
            MotionEvent evento,
            int indice) {

        int id = evento.getPointerId(indice);

        float x = obterXNaTela(evento, indice);

        float y = obterYNaTela(evento, indice);

        int controle = identificarControle(x, y);

        if (controle == 0) {

            if (view == controleEsquerdo) {

                controle = 1;

            } else if (view == controleDireito) {

                controle = 2;
            }
        }

        if (controle == 1 && idDedoEsquerdo == -1) {

            idDedoEsquerdo = id;

            esquerdoPressionado = true;

            posicaoDedoEsquerdo = y;

            Log.d(TAG, "Dedo no controle esquerdo. ID: " + id);

        } else if (controle == 2 &&
                idDedoDireito == -1) {

            idDedoDireito = id;

            direitoPressionado = true;

            posicaoDedoDireito = y;

            Log.d(TAG, "Dedo no controle direito. ID: " + id);
        }
    }

    // =====================================================
    // ATUALIZA POSIÇÕES DOS DEDOS
    // =====================================================

    private void atualizarPosicoesDedos(
            MotionEvent evento) {

        for (int i = 0; i < evento.getPointerCount(); i++) {

            int id = evento.getPointerId(i);

            float y = obterYNaTela(evento, i);

            if (id == idDedoEsquerdo) {

                posicaoDedoEsquerdo = y;

            } else if (id == idDedoDireito) {

                posicaoDedoDireito = y;
            }
        }
    }

    // =====================================================
    // LIBERA UM DEDO
    // =====================================================

    private void liberarPonteiro(int id) {

        if (id == idDedoEsquerdo) {

            idDedoEsquerdo = -1;

            esquerdoPressionado = false;

            Log.d(TAG, "Controle esquerdo liberado.");
        }

        if (id == idDedoDireito) {

            idDedoDireito = -1;

            direitoPressionado = false;

            Log.d(TAG, "Controle direito liberado.");
        }
    }

    // =====================================================
    // LIBERA TODOS OS DEDOS
    // =====================================================

    private void liberarTodosOsDedos() {

        idDedoEsquerdo = -1;
        idDedoDireito = -1;

        esquerdoPressionado = false;
        direitoPressionado = false;
    }

    // =====================================================
    // INICIA CONTROLE MANUAL
    // =====================================================

    private void verificarControleManual() {

        if (!esquerdoPressionado &&
                !direitoPressionado) {

            return;
        }

        if (!listener.possuiGif()) {
            return;
        }

        if (!controleManualAtivo) {

            controleManualAtivo = true;

            Log.d(TAG, "CONTROLE MANUAL INICIADO.");

            listener.onControleManualIniciado();
        }
    }

    // =====================================================
    // ATUALIZA POSIÇÃO DOS CONTROLES
    // =====================================================

    private void atualizarControleManual() {

        if (!controleManualAtivo ||
                (!esquerdoPressionado &&
                        !direitoPressionado) ||
                !listener.possuiGif()) {

            return;
        }

        int altura = areaGif.getHeight();

        if (altura <= 0) return;

        int[] localizacao = new int[2];

        areaGif.getLocationOnScreen(localizacao);

        float posicaoMedia;

        if (esquerdoPressionado &&
                direitoPressionado) {

            posicaoMedia =
                    (posicaoDedoEsquerdo +
                            posicaoDedoDireito) / 2f;

        } else if (esquerdoPressionado) {

            posicaoMedia = posicaoDedoEsquerdo;

        } else {

            posicaoMedia = posicaoDedoDireito;
        }

        float posicaoLocal =
                posicaoMedia - localizacao[1];

        float metadeBotao =
                controleEsquerdo.getHeight() / 2f;

        float limiteInferior = metadeBotao;

        float limiteSuperior =
                altura - metadeBotao;

        if (limiteSuperior <= limiteInferior) {
            return;
        }

        posicaoLocal = Math.max(
                limiteInferior,
                Math.min(limiteSuperior, posicaoLocal)
        );

        float progresso =
                (posicaoLocal - metadeBotao) /
                        (altura - 2f * metadeBotao);

        progresso = Math.max(
                0f,
                Math.min(1f, progresso)
        );

        int centroY = (int) posicaoLocal;

        if (esquerdoPressionado &&
                direitoPressionado) {

            moverBotao(
                    controleEsquerdo,
                    centroY,
                    true
            );

            moverBotao(
                    controleDireito,
                    centroY,
                    false
            );

        } else if (esquerdoPressionado) {

            moverBotao(
                    controleEsquerdo,
                    centroY,
                    true
            );

        } else {

            moverBotao(
                    controleDireito,
                    centroY,
                    false
            );
        }

        listener.onProgressoManual(progresso);
    }

    // =====================================================
    // MOVIMENTAÇÃO DOS BOTÕES
    // =====================================================

    public void moverControles(float progresso) {

        final float progressoFinal = Math.max(
                0f,
                Math.min(1f, progresso)
        );

        areaGif.post(() -> {

            int altura = areaGif.getHeight();

            int alturaBotao =
                    controleEsquerdo.getHeight();

            if (altura <= 0 || alturaBotao <= 0) {
                return;
            }

            float metadeBotao = alturaBotao / 2f;

            int centroY = (int) (
                    metadeBotao +
                            progressoFinal *
                                    (altura - 2f * metadeBotao)
            );

            moverBotao(
                    controleEsquerdo,
                    centroY,
                    true
            );

            moverBotao(
                    controleDireito,
                    centroY,
                    false
            );
        });
    }

    private void moverBotao(
            TextView botao,
            int centroY,
            boolean esquerdo) {

        if (botao == null ||
                botao.getLayoutParams() == null) {

            return;
        }

        FrameLayout.LayoutParams parametros =
                (FrameLayout.LayoutParams)
                        botao.getLayoutParams();

        parametros.gravity =
                Gravity.TOP |
                        (esquerdo
                                ? Gravity.START
                                : Gravity.END);

        parametros.topMargin = Math.max(
                0,
                centroY - botao.getHeight() / 2
        );

        botao.setLayoutParams(parametros);
    }

    // =====================================================
    // LIBERAÇÃO DO CONTROLE MANUAL
    // =====================================================

    private void verificarLiberacao() {

        if (esquerdoPressionado ||
                direitoPressionado) {

            atualizarControleManual();

            return;
        }

        boolean estavaAtivo = controleManualAtivo;

        controleManualAtivo = false;

        if (estavaAtivo) {

            Log.d(TAG, "CONTROLE MANUAL ENCERRADO.");

            listener.onControleManualLiberado();
        }
    }

    // =====================================================
    // CONSULTA DE ESTADO
    // =====================================================

    public boolean isEsquerdoPressionado() {
        return esquerdoPressionado;
    }

    public boolean isDireitoPressionado() {
        return direitoPressionado;
    }

    public boolean isControleManualAtivo() {
        return controleManualAtivo;
    }

    // =====================================================
    // LIMPEZA
    // =====================================================

    public void limpar() {

        liberarTodosOsDedos();

        controleManualAtivo = false;
    }
}

package com.example.rpghqconspiration;

import java.io.File;

/**
 * Responsável pelo acesso aos arquivos GIF locais
 * e pela obtenção de GIFs no Google Drive.
 *
 * Não manipula Views, Activities ou animações.
 */
public final class GifRepository {

    private final File pastaGifs;

    public GifRepository(File pastaGifs) {

        if (pastaGifs == null) {
            throw new IllegalArgumentException(
                    "A pasta de GIFs não pode ser nula."
            );
        }

        this.pastaGifs = pastaGifs;

        // Garante que a pasta de GIFs exista.
        if (!this.pastaGifs.exists()) {
            boolean criada = this.pastaGifs.mkdirs();

            if (!criada && !this.pastaGifs.exists()) {
                throw new IllegalStateException(
                        "Não foi possível criar a pasta de GIFs."
                );
            }
        }

        if (!this.pastaGifs.isDirectory()) {
            throw new IllegalStateException(
                    "O caminho informado não é uma pasta válida."
            );
        }
    }

    /**
     * Procura um GIF já armazenado localmente.
     *
     * Retorna null quando o arquivo ainda não existe
     * ou está vazio.
     */
    public File localizar(String link) throws Exception {

        String id = obterId(link);

        File arquivo = new File(
                pastaGifs,
                id + ".gif"
        );

        if (arquivo.isFile() && arquivo.length() > 0) {
            return arquivo;
        }

        return null;
    }

    /**
     * Baixa um GIF do Google Drive.
     */
    public File baixar(String link) throws Exception {

        String id = obterId(link);

        File arquivo = GoogleDriveDownloader.baixar(
                id,
                pastaGifs
        );

        validarArquivo(arquivo);

        return arquivo;
    }

    /**
     * Obtém um GIF local ou realiza seu download
     * quando ele ainda não está armazenado.
     */
    public File obter(String link) throws Exception {

        // Primeiro verifica se o GIF já existe.
        File arquivo = localizar(link);

        if (arquivo != null) {
            return arquivo;
        }

        // Se não existir, realiza o download.
        return baixar(link);
    }

    /**
     * Valida o arquivo retornado pelo downloader.
     */
    private void validarArquivo(File arquivo) {

        if (arquivo == null ||
                !arquivo.isFile() ||
                arquivo.length() == 0) {

            throw new IllegalStateException(
                    "Não foi possível obter um arquivo GIF válido."
            );
        }
    }

    /**
     * Valida e extrai o ID do arquivo.
     */
    private String obterId(String link) {

        if (link == null || link.trim().isEmpty()) {
            throw new IllegalArgumentException(
                    "O link do Google Drive não pode ser vazio."
            );
        }

        String id = GoogleDriveUtils.extrairId(link);

        if (id == null || id.trim().isEmpty()) {
            throw new IllegalArgumentException(
                    "Link inválido do Google Drive."
            );
        }

        return id;
    }
}

package com.example.rpghqconspiration;

import android.graphics.Movie;
import android.os.Handler;
import android.os.SystemClock;

/**
 * Responsável pelo controle temporal da reprodução dos GIFs.
 *
 * Controla:
 * - início e pausa da reprodução;
 * - posição atual da animação;
 * - atualização periódica dos quadros.
 *
 * Não manipula Views nem decide como o quadro será desenhado.
 */
public final class GerenciadorReproducaoGif {

    public interface Listener {

        /**
         * Informa se a reprodução pode ser iniciada.
         */
        boolean podeIniciar();

        /**
         * Solicita à MainActivity que desenhe o quadro atual.
         */
        void atualizarQuadro();
    }

    private final AnimationTicker animationTicker;
    private final Listener listener;

    private long posicaoMs = 0;
    private long inicioReproducao = 0;

    private boolean reproduzindo = false;

    public GerenciadorReproducaoGif(
            Handler handler,
            long intervalo,
            Listener listener) {

        if (handler == null || listener == null) {
            throw new IllegalArgumentException(
                    "Handler e Listener são obrigatórios."
            );
        }

        this.listener = listener;

        animationTicker = new AnimationTicker(
                handler,
                this::atualizar,
                intervalo
        );
    }

    private void atualizar() {

        if (!reproduzindo) {
            return;
        }

        long agora = SystemClock.uptimeMillis();

        long decorrido = agora - inicioReproducao;

        posicaoMs += decorrido;

        inicioReproducao = agora;

        listener.atualizarQuadro();
    }

    public void iniciar(Movie movie) {

        if (movie == null ||
                reproduzindo ||
                !listener.podeIniciar()) {

            return;
        }

        reproduzindo = true;

        inicioReproducao =
                SystemClock.uptimeMillis();

        animationTicker.iniciar();
    }

    public void pausar(Movie movie) {

        if (movie == null) {
            return;
        }

        if (reproduzindo) {

            int duracao = obterDuracao(movie);

            long agora = SystemClock.uptimeMillis();

            long decorrido = agora - inicioReproducao;

            posicaoMs = (
                    posicaoMs + decorrido
            ) % duracao;
        }

        reproduzindo = false;

        animationTicker.parar();
    }

    public int obterDuracao(Movie movie) {

        if (movie == null) {
            return 1000;
        }

        int duracao = movie.duration();

        return duracao > 0 ? duracao : 1000;
    }

    public long getPosicaoMs() {
        return posicaoMs;
    }

    public void definirPosicao(long posicaoMs) {

        this.posicaoMs = Math.max(0, posicaoMs);
    }

    public boolean isReproduzindo() {
        return reproduzindo;
    }

    public void reiniciar() {

        animationTicker.parar();

        reproduzindo = false;

        posicaoMs = 0;

        inicioReproducao = 0;
    }

    public void encerrar() {

        animationTicker.parar();

        reproduzindo = false;

        inicioReproducao = 0;
    }
}

/**
 * Responsável pelo processamento e envio dos comandos
 * de controle remoto entre os usuários A e B.
 *
 * Não manipula Views, GIFs ou movimentos diretamente.

public final class GerenciadorControleRemoto {

    private static final String TAG =
            "GerenciadorControleRemoto";

    public interface Listener {

        void onControleRecebido(float progresso);

        void onControleLiberado();
    }

    private final Listener listener;

    public GerenciadorControleRemoto(Listener listener) {

        if (listener == null) {
            throw new IllegalArgumentException(
                    "Listener não pode ser nulo."
            );
        }

        this.listener = listener;
    }

    /**
     * Interpreta um comando recebido do parceiro.
     */
    public void processar(JSONObject dados) {

        if (dados == null) {
            return;
        }

        boolean ativo = dados.optBoolean(
                "active",
                false
        );

        if (!ativo) {

            Log.d(
                    TAG,
                    "Parceiro liberou os controles."
            );

            listener.onControleLiberado();

            return;
        }

        float progresso = (float) dados.optDouble(
                "progress",
                0
        );

        if (Float.isNaN(progresso) ||
                Float.isInfinite(progresso)) {

            Log.w(
                    TAG,
                    "Progresso recebido inválido."
            );

            return;
        }

        progresso = Math.max(
                0f,
                Math.min(1f, progresso)
        );

        listener.onControleRecebido(progresso);
    }

    /**
     * Encaminha um comando ao parceiro.
     */
    public void enviar(
            ServicoConexao servicoConexao,
            boolean modoIndividual,
            boolean ativo,
            float progresso) {

        if (modoIndividual) {
            return;
        }

        if (servicoConexao == null) {

            Log.w(
                    TAG,
                    "Comando não enviado: serviço inexistente."
            );

            return;
        }

        if (!servicoConexao.estaConectado()) {

            Log.w(
                    TAG,
                    "Comando não enviado: conexão indisponível."
            );

            return;
        }

        if (Float.isNaN(progresso) ||
                Float.isInfinite(progresso)) {

            return;
        }

        progresso = Math.max(
                0f,
                Math.min(1f, progresso)
        );

        servicoConexao.enviarControle(
                ativo,
                progresso
        );

        Log.d(
                TAG,
                "Controle enviado. Ativo: " +
                        ativo +
                        " | Progresso: " +
                        progresso
        );
    }

        if (planilhaRepository == null ||
                executor == null ||
                handler == null ||
                listener == null) {

            throw new IllegalArgumentException(
                    "As dependências não podem ser nulas."
            );

        this.planilhaRepository = planilhaRepository;
        this.executor = executor;
        this.handler = handler;
        this.listener = listener;
    }

    // =====================================================

        executor.execute(() -> {

            try {

                Log.d(
                        TAG,
                        "Iniciando consulta à planilha."
                );

                // -----------------------------------------
                // BAIXAR CONTEÚDO CSV
                // -----------------------------------------

                String conteudo =
                        planilhaRepository.baixarConteudo();

    public int proximo(int quantidade) {

        if (quantidade <= 0) {
            indiceAtual = -1;
            return -1;
        }

        indiceAtual++;

        if (indiceAtual >= quantidade) {
            indiceAtual = 0;
        }

        return indiceAtual;
    }

    /**
     * Reinicia a navegação.
     */
    public void reiniciar() {
        indiceAtual = -1;
    }

    /**
     * Retorna o índice atualmente selecionado.
     */
    public int getIndiceAtual() {
        return indiceAtual;
    }

    /**
     * Informa se existe um GIF selecionado.
     */
    public boolean possuiSelecao() {
        return indiceAtual >= 0;
    }
}

            );

        this.handler = handler;
        this.listener = listener;
    }

                                    "O link do GIF está vazio."
                            )
                    )
            );

            return;
        }

        handler.post(listener::onCarregamentoIniciado);

        executor.execute(() -> {

            try {

                Log.d(
                        TAG,
                        "Obtendo GIF: " + link
                );

                // O repositório primeiro procura
                // o arquivo local e baixa se necessário.

                File arquivo =
                        gifRepository.obter(link);

                validarArquivo(arquivo);

                Log.d(
                        TAG,
                        "GIF obtido: " +
                                arquivo.getAbsolutePath()
                );

                handler.post(() ->
                        listener.onCarregamentoConcluido(
                                arquivo
                        )
                );

            } catch (Exception e) {

                Log.e(
                        TAG,
                        "Falha ao obter GIF.",
                        e
                );

                handler.post(() ->
                        listener.onErro(e)
                );
            }
        });
    }

    // VALIDAR ARQUIVO

    private void validarArquivo(File arquivo) {

                !arquivo.isFile() ||
                arquivo.length() <= 0) {

            throw new IllegalStateException(
                    "O arquivo GIF recebido é inválido."
            );
        }
    }
}

/**
 * Coordena o processo de exibição de um GIF.
 *
 * Responsabilidades:
 * - Interromper a reprodução anterior.
 * - Reiniciar os controles.
 * - Solicitar a decodificação do GIF.
 * - Atualizar o Movie na MainActivity.
 * - Iniciar a animação e o texto correspondente.
 *

public final class CoordenadorExibicaoGif {

    private static final String TAG = "CoordenadorExibicaoGif";

    private final FrameLayout areaGif;

    private final GerenciadorDecodificacaoGif decodificador;
    private final GerenciadorReproducaoGif reproducao;

    public interface Listener {

        boolean atividadeDisponivel();

        void definirMovie(Movie movie);

        void definirControleRemotoAtivo(boolean ativo);

        void desenharQuadro();

        void iniciarAnimacao();

        void enviarComandoControle(
                boolean ativo,
                float progresso
        );

        void atualizarVisibilidadeControles();

        void atualizarStatus(String mensagem);
    }

    public CoordenadorExibicaoGif(
            FrameLayout areaGif,
            GerenciadorDecodificacaoGif decodificador,
            GerenciadorReproducaoGif reproducao,
            GerenciadorTextoPlanilha textoPlanilha,
            GerenciadorMovimento movimento,
            GerenciadorControleMultitoque controle,
            Listener listener
    ) {

        this.areaGif = areaGif;
        this.decodificador = decodificador;
        this.reproducao = reproducao;
        this.textoPlanilha = textoPlanilha;
        this.movimento = movimento;
        this.controle = controle;
        this.listener = listener;
    }

    private boolean disponivel() {

        return !encerrado &&
                areaGif != null &&
                listener != null &&
                listener.atividadeDisponivel();
    }

    /**
     * Inicia o processo de exibição de um GIF.
     */
    public void exibir(
            File arquivo,
            String nome,
            String texto,
            int indiceExibido,
            int quantidadeTotal
    ) {

        if (!disponivel()) {
            return;
        }

        if (arquivo == null ||
                !arquivo.isFile() ||
                arquivo.length() == 0) {

            listener.atualizarStatus(
                    "Arquivo GIF inválido."
            );

            return;
        }

        final long geracaoAtual = ++geracaoExibicao;

        // 1. Interromper a reprodução anterior.

        reproducao.reiniciar();
        textoPlanilha.parar();

        // 2. Desativar o controle manual, se necessário.

        if (controle != null &&
                controle.isControleManualAtivo()) {

            listener.enviarComandoControle(false, 0f);
        }

        // 3. Limpar o estado do GIF anterior.

        listener.definirMovie(null);

        listener.definirControleRemotoAtivo(false);

        // 4. Reiniciar controles e movimento.

        movimento.reiniciarContagem();

        controle.limpar();

        listener.atualizarVisibilidadeControles();

        // 5. Reposicionar os controles.

        areaGif.post(() -> {

            if (!disponivel() ||
                    geracaoAtual != geracaoExibicao) {

                return;
            }

            controle.moverControles(0.5f);
        });

        // 6. Solicitar a decodificação do GIF.

        decodificador.decodificar(
                arquivo,
                new GerenciadorDecodificacaoGif.Listener() {

                    @Override
                    public void onDecodificacaoConcluida(
                            Movie resultado
                    ) {

                        if (!disponivel() ||
                                geracaoAtual != geracaoExibicao) {

                            return;
                        }

                        if (resultado == null) {

                            tratarErro(
                                    arquivo,
                                    "Não foi possível decodificar o GIF."
                            );

                            return;
                        }

                        // 7. Atualizar o GIF atual.

                        listener.definirMovie(resultado);

                        reproducao.reiniciar();

                        // 8. Atualizar a interface na thread principal.

                        areaGif.post(() -> {

                            if (!disponivel() ||
                                    geracaoAtual != geracaoExibicao) {

                                return;
                            }

                            listener.desenharQuadro();

                            listener.iniciarAnimacao();

                            textoPlanilha.iniciar(texto);

                            listener.atualizarStatus(
                                    nome + " | " +
                                            indiceExibido + "/" +
                                            quantidadeTotal
                            );
                        });
                    }

                    @Override
                    public void onErro(Exception erro) {

                        if (!disponivel() ||
                                geracaoAtual != geracaoExibicao) {

                            return;
                        }

                        tratarErro(
                                arquivo,
                                "Erro ao abrir GIF: " +
                                        erro.getMessage()
                        );

                        Log.e(
                                TAG,
                                "Erro ao decodificar GIF.",
                                erro
                        );
                    }
                }
        );
    }

    /**
     * Trata falhas durante a exibição.
     */
    private void tratarErro(
            File arquivo,
            String mensagem
    ) {

        reproducao.reiniciar();

        textoPlanilha.parar();

        listener.definirMovie(null);

        if (arquivo != null && arquivo.exists()) {

            arquivo.delete();
        }

        listener.atualizarStatus(mensagem);
    }

    /**
     * Invalida callbacks de uma exibição anterior.
     */
    public void invalidar() {

        geracaoExibicao++;
    }

    /**
     * Encerra o coordenador.
     */
    public void encerrar() {

        encerrado = true;

        invalidar();
    }

    public boolean estaEncerrado() {

        return encerrado;
    }
}

/**
 * Gerencia a apresentação temporizada do texto
 * associado ao GIF atual.
 *
 * Não consulta a planilha nem controla a animação.
 */
public final class GerenciadorTextoPlanilha {

    private static final long JANELA_TEXTO_MS = 7000;
    private static final long TEMPO_TEXTO_MS = 5000;
    private static final long INTERVALO_MS = 50;

    public interface Listener {
        boolean atividadeDisponivel();
        boolean possuiGif();
    }

    private final TextView texto;
    private final FrameLayout areaGif;
    private final Handler handler;
    private final Listener listener;

    private long inicioExibicao = 0;

    private boolean ativo = false;
    private boolean encerrado = false;

    private final Runnable atualizar = new Runnable() {
        @Override
        public void run() {

            if (encerrado || !ativo) {
                return;
            }

            if (!listener.atividadeDisponivel() ||
                    !listener.possuiGif()) {

                parar();
                return;
            }

            long decorrido =
                    SystemClock.uptimeMillis() - inicioExibicao;

            if (decorrido >= TEMPO_TEXTO_MS &&
                    decorrido < JANELA_TEXTO_MS) {

                texto.setVisibility(View.VISIBLE);

            } else {

                texto.setVisibility(View.GONE);
            }

            if (decorrido < JANELA_TEXTO_MS) {

                handler.postDelayed(
                        this,
                        INTERVALO_MS
                );

            } else {

                ativo = false;
                texto.setVisibility(View.GONE);
            }
        }
    };

    public GerenciadorTextoPlanilha(
            TextView texto,
            FrameLayout areaGif,
            Handler handler,
            Listener listener) {

        if (texto == null ||
                areaGif == null ||
                handler == null ||
                listener == null) {

            throw new IllegalArgumentException(
                    "Os componentes e o Listener são obrigatórios."
            );
        }

        this.texto = texto;
        this.areaGif = areaGif;
        this.handler = handler;
        this.listener = listener;
    }

    /**
     * Configura aparência e posicionamento.
     */
    public void configurar() {

        texto.setBackground(
                new GradientDrawable(
                        GradientDrawable.Orientation.TOP_BOTTOM,
                        new int[]{
                                Color.argb(250, 0, 0, 0),
                                Color.argb(220, 0, 0, 0),
                                Color.argb(140, 0, 0, 0),
                                Color.argb(0, 0, 0, 0)
                        }
                )
        );

        texto.setVisibility(View.GONE);

        areaGif.post(() -> {

            if (encerrado ||
                    !listener.atividadeDisponivel()) {
                return;
            }

            int altura = areaGif.getHeight();

            if (altura <= 0) {
                return;
            }

            FrameLayout.LayoutParams parametros =
                    (FrameLayout.LayoutParams)
                            texto.getLayoutParams();

            parametros.width =
                    FrameLayout.LayoutParams.MATCH_PARENT;

            parametros.height = altura / 4;

            parametros.gravity = Gravity.TOP;

            texto.setLayoutParams(parametros);
        });
    }

    /**
     * Inicia a apresentação do texto para o GIF atual.
     */
    public void iniciar(String conteudo) {

        if (encerrado ||
                !listener.atividadeDisponivel() ||
                !listener.possuiGif()) {
            return;
        }

        parar();

        texto.setText(conteudo == null ? "" : conteudo);

        texto.setVisibility(View.GONE);

        inicioExibicao = SystemClock.uptimeMillis();

        ativo = true;

        handler.post(atualizar);
    }

    /**
     * Interrompe a apresentação e oculta o texto.
     */
    public void parar() {

        ativo = false;

        inicioExibicao = 0;

        handler.removeCallbacks(atualizar);

        texto.setVisibility(View.GONE);
    }

    /**
     * Libera o gerenciador definitivamente.
     */
    public void encerrar() {

        parar();

        encerrado = true;
    }
}

                gerenciadorReproducao == null ||

        this.areaGif = areaGif;
        this.controleEsquerdo = controleEsquerdo;
        this.controleDireito = controleDireito;
        this.gerenciadorReproducao = gerenciadorReproducao;
        this.host = host;
        this.modoIndividual = modoIndividual;

        // -------------------------------------------------

        gerenciadorControle =

        // GERENCIADOR DE CONTROLE REMOTO

    @Override
    public void onControleRecebido(float progresso) {

        if (!podeInteragir() ||
                !host.possuiGif()) {

            return;
        }

        controleRemotoAtivo = true;

        host.pausarAnimacao();

        gerenciadorMovimento.registrarProgresso(
                progresso,
                gerenciadorControle.isControleManualAtivo(),
                controleRemotoAtivo
        );

        int duracao = host.obterDuracaoGif();

        gerenciadorReproducao.definirPosicao(
                (long) (progresso * duracao)
        );

        areaGif.post(() -> {

            if (!podeInteragir() ||
                    !host.possuiGif()) {

                return;
            }

            gerenciadorControle.moverControles(
                    progresso
            );

            host.desenharQuadro();
        });
    }

    public void onControleLiberado() {

        controleRemotoAtivo = false;

                progresso,
                true,
                controleRemotoAtivo
        );

        int duracao = host.obterDuracaoGif();

        gerenciadorReproducao.definirPosicao(
                (long) (progresso * duracao)

        );
    }

    // LIBERAÇÃO DO CONTROLE MANUAL
    // =====================================================

    @Override

    }

    public void avancarParaProximoGif() {

        if (!podeInteragir() ||

        }

        host.avancarParaProximoGif();
    }

    @Override
    public void retomarAnimacao() {

        if (!podeInteragir() ||

        }

        host.retomarAnimacao();
    }

    // =====================================================
    // VERIFICAÇÃO DE DISPONIBILIDADE
    // =====================================================

    private boolean podeInteragir() {

    }

    // =====================================================

        }

        encerrado = true;

        controleRemotoAtivo = false;

        gerenciadorControle.limpar();

        gerenciadorMovimento.reiniciarContagem();

        servicoConexao = null;

        Log.d(
                TAG,
                "Coordenador de interação encerrado."
        );
    }

package com.example.rpghqconspiration;

import android.os.Handler;
import android.util.Log;

import java.io.File;
import java.util.List;
import java.util.concurrent.ExecutorService;

/**
 * Coordena o fluxo entre:
 *
 * 1. Seleção do próximo GIF.
 * 2. Validação da posição selecionada.
 * 3. Solicitação do carregamento.
 * 4. Comunicação do resultado para a camada responsável pela interface.
 *
 * Não manipula Views diretamente.
 * Não decodifica GIFs.
 * Não controla a reprodução.
 * Não é responsável por encerrar o ExecutorService compartilhado.
 */
public final class CoordenadorFluxoGif
        implements GerenciadorCarregamentoGif.Listener {

    private static final String TAG = "CoordenadorFluxoGif";

    private final List<String> nomes;
    private final List<String> links;

    private final GerenciadorNavegacaoGif navegacao;
    private final GerenciadorCarregamentoGif carregamento;

    private final Listener listener;

    private String nomeGifSolicitado = "";

    private boolean encerrado = false;

    /**
     * Contrato de comunicação com a Activity ou outro coordenador.
     */
    public interface Listener {

        boolean atividadeDisponivel();

        boolean estaCarregando();

        void definirCarregando(boolean carregando);

        void definirNomeGifSolicitado(String nome);

        void definirBotaoProximoHabilitado(boolean habilitado);

        void atualizarStatus(String mensagem);

        void exibirGif(File arquivo, String nome);
    }

    /**
     * Constrói o fluxo de navegação e carregamento.
     *
     * As listas recebidas são mantidas como referências.
     * Portanto, quando a planilha atualizar essas listas,
     * o coordenador utilizará os dados atualizados.
     */
    public CoordenadorFluxoGif(
            List<String> nomes,
            List<String> links,
            GifRepository gifRepository,
            ExecutorService executor,
            Handler handler,
            Listener listener
    ) {

        if (nomes == null ||
                links == null ||
                gifRepository == null ||
                executor == null ||
                handler == null ||
                listener == null) {

            throw new IllegalArgumentException(
                    "As dependências não podem ser nulas."
            );
        }

        this.nomes = nomes;
        this.links = links;
        this.listener = listener;

        navegacao = new GerenciadorNavegacaoGif();

        carregamento = new GerenciadorCarregamentoGif(
                gifRepository,
                executor,
                handler,
                this
        );
    }

    /**
     * Verifica se o coordenador ainda pode executar operações.
     */
    private boolean disponivel() {

        return !encerrado &&
                listener != null &&
                listener.atividadeDisponivel();
    }

    /**
     * Solicita o próximo GIF da lista.
     *
     * A navegação é sequencial e retorna ao primeiro
     * elemento quando chega ao final.
     */
    public void carregarProximo() {

        if (!disponivel() ||
                listener.estaCarregando()) {

            return;
        }

        if (nomes.isEmpty()) {

            listener.atualizarStatus(
                    "Nenhum GIF disponível."
            );

            return;
        }

        int indiceSelecionado =
                navegacao.proximo(nomes.size());

        if (indiceSelecionado < 0) {
            return;
        }

        carregarPosicao(indiceSelecionado);
    }

    /**
     * Solicita o carregamento de uma posição específica.
     */
    public void carregarPosicao(int posicao) {

        if (!disponivel() ||
                listener.estaCarregando()) {

            return;
        }

        if (posicao < 0 ||
                posicao >= nomes.size() ||
                posicao >= links.size()) {

            Log.e(
                    TAG,
                    "Posição inválida para carregamento: " +
                            posicao
            );

            return;
        }

        nomeGifSolicitado = nomes.get(posicao);

        String link = links.get(posicao);

        listener.definirNomeGifSolicitado(
                nomeGifSolicitado
        );

        listener.atualizarStatus(
                "Carregando: " + nomeGifSolicitado
        );

        carregamento.carregar(link);
    }

    /**
     * Reinicia o índice da navegação.
     *
     * Deve ser chamado quando uma nova lista de GIFs
     * for carregada da planilha.
     */
    public void reiniciarNavegacao() {

        if (encerrado) {
            return;
        }

        navegacao.reiniciar();

        nomeGifSolicitado = "";

        listener.definirNomeGifSolicitado("");
    }

    /**
     * Retorna o índice atualmente selecionado.
     */
    public int getIndiceAtual() {

        return navegacao.getIndiceAtual();
    }

    /**
     * Informa a quantidade atual de GIFs disponíveis.
     */
    public int getQuantidade() {

        return nomes.size();
    }

    /**
     * Informa se existe um GIF selecionado.
     */
    public boolean possuiSelecao() {

        return navegacao.possuiSelecao();
    }

    /**
     * Chamado quando o GerenciadorCarregamentoGif
     * inicia a obtenção do arquivo.
     */
    @Override
    public void onCarregamentoIniciado() {

        if (!disponivel()) {
            return;
        }

        listener.definirCarregando(true);

        listener.definirBotaoProximoHabilitado(false);

        listener.atualizarStatus(
                "Carregando: " + nomeGifSolicitado
        );
    }

    /**
     * Chamado quando o arquivo GIF foi obtido.
     */
    @Override
    public void onCarregamentoConcluido(File arquivo) {

        if (!disponivel()) {
            return;
        }

        listener.definirCarregando(false);

        listener.definirBotaoProximoHabilitado(true);

        if (arquivo == null ||
                !arquivo.isFile() ||
                arquivo.length() <= 0) {

            onErro(
                    new IllegalStateException(
                            "O arquivo GIF recebido é inválido."
                    )
            );

            return;
        }

        listener.exibirGif(
                arquivo,
                nomeGifSolicitado
        );
    }

    /**
     * Chamado quando ocorre uma falha no carregamento.
     */
    @Override
    public void onErro(Exception erro) {

        if (!disponivel()) {
            return;
        }

        listener.definirCarregando(false);

        listener.definirBotaoProximoHabilitado(true);

        String mensagem = erro != null
                ? erro.getMessage()
                : "Erro desconhecido.";

        listener.atualizarStatus(
                "Erro ao obter GIF: " + mensagem
        );

        Log.e(
                TAG,
                "Erro ao carregar GIF.",
                erro
        );
    }

    /**
     * Impede novas operações e ignora callbacks posteriores.
     *
     * O ExecutorService não é encerrado aqui porque
     * pertence à camada que o criou.
     */
    public void encerrar() {

        encerrado = true;

        nomeGifSolicitado = "";
    }

    public boolean estaEncerrado() {

        return encerrado;
    }
}

package com.example.rpghqconspiration;

import android.os.Handler;
import android.util.Log;

import java.util.ArrayList;
import java.util.List;
import java.util.concurrent.ExecutorService;

/**
 * Coordena a integração entre o carregamento da planilha
 * e a disponibilização dos GIFs para navegação.
 *
 * Responsabilidades:
 * - Solicitar o carregamento da planilha.
 * - Receber os dados interpretados.
 * - Atualizar as listas compartilhadas de nomes, links e textos.
 * - Reiniciar a navegação após uma nova consulta.
 * - Iniciar o primeiro GIF quando houver dados.
 * - Comunicar o resultado à camada de interface.
 *
 * Não realiza requisições HTTP diretamente.
 * Não interpreta CSV diretamente.
 * Não manipula Views.
 * Não baixa nem decodifica GIFs.
 */
public final class CoordenadorPlanilhaGif
        implements GerenciadorCarregamentoPlanilha.Listener {

    private static final String TAG = "CoordenadorPlanilhaGif";

    private final List<String> nomes;
    private final List<String> links;
    private final List<String> textos;

    private final GerenciadorCarregamentoPlanilha carregamentoPlanilha;

    private final CoordenadorFluxoGif fluxoGif;

    private final Listener listener;

    private boolean encerrado = false;

    /**
     * Comunicação com a camada responsável pela interface.
     */
    public interface Listener {

        boolean atividadeDisponivel();

        void atualizarStatus(String mensagem);

        void definirBotaoProximoHabilitado(boolean habilitado);
    }

    public CoordenadorPlanilhaGif(
            PlanilhaRepository planilhaRepository,
            ExecutorService executor,
            Handler handler,
            List<String> nomes,
            List<String> links,
            List<String> textos,
            CoordenadorFluxoGif fluxoGif,
            Listener listener
    ) {

        if (planilhaRepository == null ||
                executor == null ||
                handler == null ||
                nomes == null ||
                links == null ||
                textos == null ||
                fluxoGif == null ||
                listener == null) {

            throw new IllegalArgumentException(
                    "As dependências não podem ser nulas."
            );
        }

        this.nomes = nomes;
        this.links = links;
        this.textos = textos;

        this.fluxoGif = fluxoGif;
        this.listener = listener;

        carregamentoPlanilha =
                new GerenciadorCarregamentoPlanilha(
                        planilhaRepository,
                        executor,
                        handler,
                        this
                );
    }

    /**
     * Verifica se o coordenador pode continuar executando.
     */
    private boolean disponivel() {

        return !encerrado &&
                listener != null &&
                listener.atividadeDisponivel();
    }

    /**
     * Solicita uma nova consulta à planilha.
     */
    public void carregar() {

        if (!disponivel()) {
            return;
        }

        listener.atualizarStatus(
                "Consultando planilha..."
        );

        listener.definirBotaoProximoHabilitado(false);

        carregamentoPlanilha.carregar();
    }

    /**
     * Recebe os dados após o carregamento e a interpretação
     * da planilha terem sido concluídos.
     */
    @Override
    public void onCarregamentoConcluido(
            GerenciadorCarregamentoPlanilha.Resultado resultado
    ) {

        if (!disponivel()) {
            return;
        }

        if (resultado == null) {

            onErro(
                    new IllegalStateException(
                            "O resultado da planilha está vazio."
                    )
            );

            return;
        }

        // Atualizar os dados compartilhados.

        nomes.clear();
        links.clear();
        textos.clear();

        nomes.addAll(resultado.getNomes());
        links.addAll(resultado.getLinks());
        textos.addAll(resultado.getTextos());

        // Reiniciar a navegação para a nova lista.

        fluxoGif.reiniciarNavegacao();

        // Verificar se existem GIFs disponíveis.

        if (resultado.estaVazio()) {

            listener.atualizarStatus(
                    "Nenhum GIF encontrado."
            );

            listener.definirBotaoProximoHabilitado(false);

            return;
        }

        listener.atualizarStatus(
                resultado.getQuantidade() +
                        " GIFs disponíveis."
        );

        listener.definirBotaoProximoHabilitado(true);

        // Iniciar a sequência pelo primeiro GIF.

        fluxoGif.carregarProximo();

        Log.d(
                TAG,
                "Dados da planilha atualizados. Registros: " +
                        resultado.getQuantidade()
        );
    }

    /**
     * Recebe erros ocorridos durante a consulta
     * ou interpretação da planilha.
     */
    @Override
    public void onErro(Exception erro) {

        if (!disponivel()) {
            return;
        }

        String mensagem = erro != null
                ? erro.getMessage()
                : "Erro desconhecido.";

        listener.atualizarStatus(
                "Erro ao consultar planilha: " + mensagem
        );

        listener.definirBotaoProximoHabilitado(false);

        Log.e(
                TAG,
                "Falha ao carregar os dados da planilha.",
                erro
        );
    }

    /**
     * Encerra o coordenador e impede o processamento
     * de novos resultados.
     *
     * O ExecutorService é compartilhado e não é encerrado aqui.
     */
    public void encerrar() {

        encerrado = true;
    }

    public boolean estaEncerrado() {

        return encerrado;
    }

    public int getQuantidadeRegistros() {

        return nomes.size();
    }

    public boolean estaVazio() {

        return nomes.isEmpty();
    }
}

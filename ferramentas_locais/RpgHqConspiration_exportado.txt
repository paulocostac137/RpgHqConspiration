                TAG,

                        usuarioBDesbloqueado +

    private void conectarServidor() {

        servicoConexao = ConexaoActivity.conexao;

        if (servicoConexao == null) {

            status.setText(
                    "ERRO: conexão com o servidor não encontrada."
            );

            Log.e(
                    TAG,
                    "ServicoConexao não foi inicializado."
            );

            atualizarVisibilidadeControles();

            return;
        }

        if (!servicoConexao.estaConectado()) {

            status.setText(
                    "ERRO: conexão com o servidor foi perdida."
            );

            Log.e(
                    TAG,
                    "A conexão compartilhada não está ativa."
            );

            atualizarVisibilidadeControles();

            return;
        }

        gerenciadorSessao.registrarCallbacks(
                servicoConexao,
                new GerenciadorSessao.Listener() {

                    @Override
                    public boolean atividadeDisponivel() {

                        return MainActivity.this.atividadeDisponivel();
                    }

                    @Override
                    public void onConectado() {

                        Log.d(
                                TAG,
                                "Conexão compartilhada ativa."
                        );

                        status.setText(
                                "Conectado ao servidor."
                        );

                        atualizarVisibilidadeControles();
                    }

                    @Override
                    public void onDesconectado(String motivo) {

                        Log.w(
                                TAG,
                                "Servidor desconectado: " + motivo
                        );

                        controleRemotoAtivo = false;

                        finalizarMovimentoSeLiberado();

                        status.setText(
                                "Servidor desconectado."
                        );
                    }

                    @Override
                    public void onMensagem(JSONObject mensagem) {

                        processarMensagemServidor(mensagem);
                    }

                    @Override
                    public void onErro(String erro) {

                        Log.e(
                                TAG,
                                "Erro de conexão: " + erro
                        );

                        status.setText(
                                "Erro de conexão: " + erro
                        );
                    }
                }
        );

        Log.d(
                TAG,
                "MainActivity reutilizando a conexão existente."
        );

        status.setText(
                "Conectado. Aguardando comandos..."
        );

        atualizarVisibilidadeControles();
    }

    private void processarControleRemoto(JSONObject dados) {

        boolean ativo = dados.optBoolean("active", false);

        if (!ativo) {

            controleRemotoAtivo = false;

            Log.d(
                    TAG,
                    "Parceiro liberou os controles."
            );

            finalizarMovimentoSeLiberado();

            return;
        }

        if (movie == null) {
            return;
        }

        float progressoRecebido =
                (float) dados.optDouble("progress", 0);

        if (Float.isNaN(progressoRecebido) ||
                Float.isInfinite(progressoRecebido)) {

            return;
        }

        progressoRecebido = Math.max(
                0f,
                Math.min(1f, progressoRecebido)
        );

        controleRemotoAtivo = true;

        pausarAnimacao();

        gerenciadorMovimento.registrarProgresso(
                progressoRecebido,
                controleManualAtivo,
                controleRemotoAtivo
        );

        int duracao = obterDuracaoGif();

        posicaoMs = (long) (
                progressoRecebido * duracao
        );

        final float progressoFinal = progressoRecebido;

        areaGif.post(() -> {

            if (!atividadeDisponivel() || movie == null) {
                return;
            }

            int altura = areaGif.getHeight();

            int alturaBotao =
                    controleEsquerdo.getHeight();

            if (altura <= 0 || alturaBotao <= 0) {
                return;
            }

            float metadeBotao = alturaBotao / 2f;

            int centroY = (int) (
                    metadeBotao +
                            progressoFinal *
                                    (altura - 2f * metadeBotao)
            );

            moverBotao(
                    controleEsquerdo,
                    centroY,
                    true
            );

            moverBotao(
                    controleDireito,
                    centroY,
                    false
            );

            desenharQuadro();
        });
    }

        );
    }

    private int obterDuracaoGif() {

        if (movie == null) {

        return duracao > 0 ? duracao : 1000;

    // =====================================================
    // ATUALIZAÇÃO AUTOMÁTICA DA ANIMAÇÃO
    // =====================================================

    private void atualizarQuadroAnimacao() {

        if (!atividadeDisponivel() ||
                movie == null ||
                !reproduzindo) {

            return;
        }

        int duracao = obterDuracaoGif();

        long agora = SystemClock.uptimeMillis();

        long decorrido = agora - inicioReproducao;

        posicaoMs = (
                posicaoMs + decorrido
        ) % duracao;

        inicioReproducao = agora;

        desenharQuadro();
    }

    private void configurarTextoPlanilha() {

        textoPlanilha.setBackground(
                new GradientDrawable(
                        GradientDrawable.Orientation.TOP_BOTTOM,
                        new int[]{
                                Color.argb(250, 0, 0, 0),
                                Color.argb(220, 0, 0, 0),

                                Color.argb(0, 0, 0, 0)
                        }

        );

        textoPlanilha.setVisibility(View.GONE);

        areaGif.post(() -> {

            if (!atividadeDisponivel()) return;

            int altura = areaGif.getHeight();

            if (altura <= 0) return;

            FrameLayout.LayoutParams parametros =
                    (FrameLayout.LayoutParams)
                            textoPlanilha.getLayoutParams();

            parametros.width =
                    FrameLayout.LayoutParams.MATCH_PARENT;

            parametros.height = altura / 4;

            parametros.gravity = Gravity.TOP;

            textoPlanilha.setLayoutParams(parametros);
        });
    }

    private final Runnable atualizarTextoPlanilha =
            new Runnable() {

                @Override
                public void run() {

                    if (!atividadeDisponivel()) {
                        return;
                    }

                    if (movie == null || inicioExibicaoGif == 0) {

                        textoPlanilha.setVisibility(View.GONE);

                        return;
                    }

                    long decorrido =
                            SystemClock.uptimeMillis() -
                                    inicioExibicaoGif;

                    if (decorrido >= TEMPO_TEXTO_MS &&
                            decorrido < JANELA_TEXTO_MS) {

                        textoPlanilha.setVisibility(View.VISIBLE);

                    } else {

                        textoPlanilha.setVisibility(View.GONE);
                    }

                    if (decorrido < JANELA_TEXTO_MS) {

                        handler.postDelayed(
                                this,
                                INTERVALO_ATUALIZACAO_MS
                        );
                    }
                }
            };

    private void configurarControles() {

        areaGif.setMotionEventSplittingEnabled(false);

        View.OnTouchListener listenerMultitoque =
                (view, evento) -> {

                    processarEventoMultitoque(
                            view,

                    );

                listenerMultitoque
        );

    // =====================================================
    // PROCESSAMENTO MULTITOQUE
    // =====================================================

    private void processarEventoMultitoque(
            View view,
            MotionEvent evento) {

        int acao = evento.getActionMasked();

        int indice = evento.getActionIndex();

        switch (acao) {

            case MotionEvent.ACTION_DOWN:

                atribuirPonteiro(
                        view,
                        evento,
                        indice
                );

                verificarControleManual();

                atualizarControleManual();

                break;

            case MotionEvent.ACTION_POINTER_DOWN:

                atribuirPonteiro(
                        view,
                        evento,
                        indice
                );

                verificarControleManual();

                atualizarControleManual();

                break;

            case MotionEvent.ACTION_MOVE:

                atualizarPosicoesDedos(evento);

                atualizarControleManual();

                break;

            case MotionEvent.ACTION_POINTER_UP:

                atualizarPosicoesDedos(evento);

                int id =
                        evento.getPointerId(indice);

                liberarPonteiro(id);

                verificarLiberacao();

                break;

            case MotionEvent.ACTION_UP:

            case MotionEvent.ACTION_CANCEL:

                liberarTodosOsDedos();

                verificarLiberacao();

                break;
        }
    }

    // =====================================================
    // CONVERTE COORDENADAS PARA A TELA
    // =====================================================

    private float obterXNaTela(
            MotionEvent evento,
            int indice) {

        float deslocamentoX =
                evento.getRawX() - evento.getX(0);

        return evento.getX(indice) + deslocamentoX;
    }

    private float obterYNaTela(
            MotionEvent evento,
            int indice) {

        float deslocamentoY =
                evento.getRawY() - evento.getY(0);

        return evento.getY(indice) + deslocamentoY;
    }

    // =====================================================
    // IDENTIFICA QUAL CONTROLE RECEBEU O DEDO
    // =====================================================

    private int identificarControle(
            float x,
            float y) {

        int[] localizacao = new int[2];

        controleEsquerdo.getLocationOnScreen(
                localizacao
        );

        if (x >= localizacao[0] &&
                x <= localizacao[0] +
                        controleEsquerdo.getWidth() &&
                y >= localizacao[1] &&
                y <= localizacao[1] +
                        controleEsquerdo.getHeight()) {

            return 1;
        }

        controleDireito.getLocationOnScreen(
                localizacao
        );

        if (x >= localizacao[0] &&
                x <= localizacao[0] +
                        controleDireito.getWidth() &&
                y >= localizacao[1] &&
                y <= localizacao[1] +
                        controleDireito.getHeight()) {

            return 2;
        }

        return 0;
    }

    // =====================================================
    // ATRIBUI UM DEDO A UM CONTROLE
    // =====================================================

    private void atribuirPonteiro(
            View view,
            MotionEvent evento,
            int indice) {

        int id = evento.getPointerId(indice);

        float x = obterXNaTela(evento, indice);

        float y = obterYNaTela(evento, indice);

        int controle = identificarControle(x, y);

        if (controle == 0) {

            if (view == controleEsquerdo) {
                controle = 1;

            } else if (view == controleDireito) {
                controle = 2;
            }
        }

        if (controle == 1 && idDedoEsquerdo == -1) {

            idDedoEsquerdo = id;

            esquerdoPressionado = true;

            posicaoDedoEsquerdo = y;

            Log.d(
                    TAG,
                    "Dedo no controle esquerdo. ID: " + id
            );

        } else if (controle == 2 &&
                idDedoDireito == -1) {

            idDedoDireito = id;

            direitoPressionado = true;

            posicaoDedoDireito = y;

            Log.d(
                    TAG,
                    "Dedo no controle direito. ID: " + id
            );
        }
    }

    // =====================================================
    // ATUALIZA POSIÇÕES DOS DEDOS
    // =====================================================

    private void atualizarPosicoesDedos(
            MotionEvent evento) {

        for (int i = 0; i < evento.getPointerCount(); i++) {

            int id = evento.getPointerId(i);

            float y = obterYNaTela(evento, i);

            if (id == idDedoEsquerdo) {

                posicaoDedoEsquerdo = y;

            } else if (id == idDedoDireito) {

                posicaoDedoDireito = y;
            }
        }
    }

    // =====================================================
    // LIBERA UM DEDO
    // =====================================================

    private void liberarPonteiro(int id) {

        if (id == idDedoEsquerdo) {

            idDedoEsquerdo = -1;

            esquerdoPressionado = false;

            Log.d(
                    TAG,
                    "Controle esquerdo liberado."
            );
        }

        if (id == idDedoDireito) {

            idDedoDireito = -1;

            direitoPressionado = false;

            Log.d(
                    TAG,
                    "Controle direito liberado."
            );
        }
    }

    private void liberarTodosOsDedos() {

        idDedoEsquerdo = -1;
        idDedoDireito = -1;

        esquerdoPressionado = false;
        direitoPressionado = false;
    }

    // =====================================================
    // INICIA CONTROLE MANUAL
    // =====================================================

    private void verificarControleManual() {

        if (!esquerdoPressionado &&
                !direitoPressionado) {

            return;
        }

        if (movie == null) {
            return;
        }

        if (!controleManualAtivo) {

            controleManualAtivo = true;

            Log.d(
                    TAG,
                    "CONTROLE MANUAL INICIADO."
            );

            pausarAnimacao();
        }
    }

    // =====================================================
    // ATUALIZA POSIÇÃO DOS CONTROLES
    // =====================================================

    private void atualizarControleManual() {

        if (!controleManualAtivo ||
                (!esquerdoPressionado &&
                        !direitoPressionado) ||
                movie == null) {

            return;
        }

        int altura = areaGif.getHeight();

        if (altura <= 0) return;

        int[] localizacao = new int[2];

        areaGif.getLocationOnScreen(localizacao);

        float posicaoMedia;

        if (esquerdoPressionado &&
                direitoPressionado) {

            posicaoMedia =
                    (posicaoDedoEsquerdo +
                            posicaoDedoDireito) / 2f;

        } else if (esquerdoPressionado) {

            posicaoMedia = posicaoDedoEsquerdo;

        } else {

            posicaoMedia = posicaoDedoDireito;
        }

        float posicaoLocal =
                posicaoMedia - localizacao[1];

        float metadeBotao =
                controleEsquerdo.getHeight() / 2f;

        float limiteInferior = metadeBotao;

        float limiteSuperior =
                altura - metadeBotao;

        if (limiteSuperior <= limiteInferior) {
            return;
        }

        posicaoLocal = Math.max(
                limiteInferior,
                Math.min(limiteSuperior, posicaoLocal)
        );

        float progresso =
                (posicaoLocal - metadeBotao) /
                        (altura - 2f * metadeBotao);

        progresso = Math.max(
                0f,
                Math.min(1f, progresso)
        );

        // O GerenciadorMovimento recebe o progresso manual.

        gerenciadorMovimento.registrarProgresso(
                progresso,
                controleManualAtivo,
                controleRemotoAtivo
        );

        int duracao = obterDuracaoGif();

        posicaoMs = (long) (
                progresso * duracao
        );

        int centroY = (int) posicaoLocal;

        if (esquerdoPressionado &&
                direitoPressionado) {

            moverBotao(
                    controleEsquerdo,
                    centroY,
                    true
            );

            moverBotao(
                    controleDireito,
                    centroY,
                    false
            );

        } else if (esquerdoPressionado) {

            moverBotao(
                    controleEsquerdo,
                    centroY,
                    true
            );

        } else {

            moverBotao(
                    controleDireito,
                    centroY,
                    false
            );
        }

        desenharQuadro();

        enviarComandoControle(
                true,
                progresso
        );
    }

    // =====================================================
    // MOVIMENTAÇÃO DOS BOTÕES
    // =====================================================

    private void moverBotao(
            TextView botao,
            int centroY,
            boolean esquerdo) {

        if (botao == null ||
                botao.getLayoutParams() == null) {

            return;
        }

        FrameLayout.LayoutParams parametros =
                (FrameLayout.LayoutParams)
                        botao.getLayoutParams();

        parametros.gravity =
                Gravity.TOP |
                        (esquerdo
                                ? Gravity.START
                                : Gravity.END);

        parametros.topMargin = Math.max(
                0,
                centroY - botao.getHeight() / 2
        );

        botao.setLayoutParams(parametros);
    }

    // =====================================================
    // LIBERAÇÃO DO CONTROLE MANUAL
    // =====================================================

    private void verificarLiberacao() {

        if (esquerdoPressionado ||
                direitoPressionado) {

            atualizarControleManual();

            return;
        }

        boolean estavaAtivo = controleManualAtivo;

        controleManualAtivo = false;

        if (estavaAtivo) {

            Log.d(
                    TAG,
                    "CONTROLE MANUAL ENCERRADO."
            );

            enviarComandoControle(false, 0f);
        }

        finalizarMovimentoSeLiberado();
    }

    // =====================================================
    // PAUSAR ANIMAÇÃO
    // =====================================================

    private void pausarAnimacao() {

        if (movie == null) return;

        if (reproduzindo) {

            int duracao = obterDuracaoGif();

            long decorrido =
                    SystemClock.uptimeMillis() -
                            inicioReproducao;

            posicaoMs = (
                    posicaoMs + decorrido
            ) % duracao;
        }

        reproduzindo = false;

        animationTicker.parar();
    }

    // =====================================================
    // RETOMAR ANIMAÇÃO
    // =====================================================

    private void iniciarAnimacao() {

        if (movie == null ||
                reproduzindo ||
                controleManualAtivo ||
                controleRemotoAtivo ||
                !atividadeDisponivel()) {

            return;
        }

        reproduzindo = true;

        inicioReproducao =
                SystemClock.uptimeMillis();

        animationTicker.iniciar();
    }

    // =====================================================
    // DESENHO DO QUADRO
    // =====================================================

    private void desenharQuadro() {

        if (!atividadeDisponivel() ||
                movie == null ||
                renderizadorGif == null) {

            return;
        }

        renderizadorGif.desenhar(
                movie,
                posicaoMs,
                pincel
        );
    }

    // =====================================================
    // CONSULTA À PLANILHA
    // =====================================================

    private void carregarPlanilha() {

        status.setText(
                "Consultando Google Planilhas..."
        );

        botaoProximo.setEnabled(false);

        executor.execute(() -> {

            try {

                String conteudo =
                        planilhaRepository.baixarConteudo();

                PlanilhaParser.preencherListas(
                        conteudo,
                        nomes,
                        links,
                        textosPlanilha
                );

                handler.post(() -> {

                    if (!atividadeDisponivel()) return;

                    indiceAtual = -1;

                    if (nomes.isEmpty()) {

                        status.setText(
                                "Nenhum GIF encontrado."
                        );

                        botaoProximo.setEnabled(false);

                    } else {

                        status.setText(
                                nomes.size() +
                                        " GIFs disponíveis."
                        );

                        botaoProximo.setEnabled(true);

                        proximoGif();
                    }
                });

            } catch (Exception e) {

                Log.e(
                        TAG,
                        "Erro na planilha.",
                        e
                );

                handler.post(() -> {

                    if (!atividadeDisponivel()) return;

                    status.setText(
                            "Erro ao consultar planilha: " +
                                    e.getMessage()
                    );

                    botaoProximo.setEnabled(false);
                });
            }
        });
    }

    // =====================================================
    // PRÓXIMO GIF
    // =====================================================

    private void proximoGif() {

        if (!atividadeDisponivel() || carregando) {
            return;
        }

        if (nomes.isEmpty()) {

            status.setText(
                    "Nenhum GIF disponível."
            );

            return;
        }

        indiceAtual++;

        if (indiceAtual >= nomes.size()) {
            indiceAtual = 0;
        }

        baixarEExibir(indiceAtual);
    }

    // =====================================================
    // DOWNLOAD DO GIF PRINCIPAL
    // =====================================================

    private void baixarEExibir(int posicao) {

        if (!atividadeDisponivel()) return;

        if (posicao < 0 || posicao >= nomes.size()) {
            return;
        }

        String nome = nomes.get(posicao);

        String link = links.get(posicao);

        String id = GoogleDriveUtils.extrairId(link);

        if (id == null) {

            status.setText(
                    "Link inválido: " + nome
            );

            return;
        }

        File arquivo = new File(
                pastaGifs,
                id + ".gif"
        );

        if (arquivo.exists() &&
                arquivo.length() > 0) {

            exibirGif(arquivo, nome);

            return;
        }

        carregando = true;

        botaoProximo.setEnabled(false);

        status.setText(
                "Baixando: " + nome
        );

        executor.execute(() -> {

            try {

                File arquivoBaixado =
                        GoogleDriveDownloader.baixar(
                                id,
                                pastaGifs
                        );

                handler.post(() -> {

                    if (!atividadeDisponivel()) return;

                    carregando = false;

                    botaoProximo.setEnabled(true);

                    exibirGif(arquivoBaixado, nome);
                });

            } catch (Exception e) {

                Log.e(
                        TAG,
                        "Erro no download do GIF.",
                        e
                );

                handler.post(() -> {

                    if (!atividadeDisponivel()) return;

                    carregando = false;

                    botaoProximo.setEnabled(true);

                    status.setText(
                            "Erro no download: " +
                                    e.getMessage()
                    );
                });
            }
        });
    }

    // =====================================================
    // EXIBIÇÃO DO GIF PRINCIPAL
    // =====================================================

    private void exibirGif(
            File arquivo,
            String nome) {

        if (!atividadeDisponivel()) return;

        long geracaoAtual = ++geracaoExibicao;

        animationTicker.parar();

        handler.removeCallbacks(atualizarTextoPlanilha);

        reproduzindo = false;

        if (controleManualAtivo) {
            enviarComandoControle(false, 0f);
        }

        movie = null;

        posicaoMs = 0;

        inicioExibicaoGif = 0;

        textoPlanilha.setVisibility(View.GONE);

        controleManualAtivo = false;

        controleRemotoAtivo = false;

        gerenciadorMovimento.reiniciarContagem();

        limparIdentificacaoDedos();

        atualizarVisibilidadeControles();

        areaGif.post(() -> {

            if (!atividadeDisponivel() ||
                    geracaoAtual != geracaoExibicao) {

                return;
            }

            int centro = areaGif.getHeight() / 2;

            moverBotao(
                    controleEsquerdo,
                    centro,
                    true
            );

            moverBotao(
                    controleDireito,
                    centro,
                    false
            );
        });

        executor.execute(() -> {

            try {

                Movie resultado = GifDecoder.carregar(arquivo);

                handler.post(() -> {

                    if (!atividadeDisponivel() ||
                            geracaoAtual != geracaoExibicao) {

                        return;
                    }

                    movie = resultado;

                    posicaoMs = 0;

                    String texto = "";

                    if (indiceAtual >= 0 &&
                            indiceAtual < textosPlanilha.size()) {

                        texto = textosPlanilha.get(indiceAtual);
                    }

                    textoPlanilha.setText(texto);

                    textoPlanilha.setVisibility(View.GONE);

                    areaGif.post(() -> {

                        if (!atividadeDisponivel() ||
                                geracaoAtual != geracaoExibicao) {

                            return;
                        }

                        desenharQuadro();

                        iniciarAnimacao();

                        inicioExibicaoGif =
                                SystemClock.uptimeMillis();

                        handler.removeCallbacks(
                                atualizarTextoPlanilha
                        );

                        handler.post(
                                atualizarTextoPlanilha
                        );
                    });

                    status.setText(
                            nome + " | " +
                                    (indiceAtual + 1) +
                                    "/" + nomes.size()
                    );
                });

            } catch (Exception e) {

                Log.e(
                        TAG,
                        "Erro ao abrir GIF.",
                        e
                );

                handler.post(() -> {

                    if (!atividadeDisponivel() ||
                            geracaoAtual != geracaoExibicao) {

                        return;
                    }

                    if (arquivo.exists()) {
                        arquivo.delete();
                    }

                    textoPlanilha.setVisibility(View.GONE);

                    status.setText(
                            "Erro ao abrir GIF: " +
                                    e.getMessage()
                    );
                });
            }
        });
    }

    // =====================================================
    // ENCERRAMENTO
    // =====================================================

    @Override
    protected void onDestroy() {

        atividadeEncerrada = true;

        geracaoExibicao++;

        if (animationTicker != null) {

            animationTicker.parar();

            animationTicker = null;
        }

        handler.removeCallbacksAndMessages(null);

        reproduzindo = false;

        movie = null;

        controleManualAtivo = false;

        controleRemotoAtivo = false;

        gerenciadorMovimento.reiniciarContagem();

        limparIdentificacaoDedos();

        /*
         * O GerenciadorSessao remove somente o callback.
         *
         * A conexão WebSocket continua pertencendo
         * à ConexaoActivity.
         */

        gerenciadorSessao.removerCallback();

        if (renderizadorGif != null) {

            renderizadorGif.liberar();

            renderizadorGif = null;
        }

        executor.shutdownNow();

        super.onDestroy();
    }
}

    private void iniciarConexao() {

        if (conexao != null && conexao.estaConectado()) {
            statusConexao.setText(
                    "Você já está conectado ao servidor."
            );
            return;
        }

        String codigo = campoCodigoPar.getText()
                .toString()
                .trim();

        if (codigo.isEmpty()) {
            campoCodigoPar.setError("Digite o código da sala.");
            campoCodigoPar.requestFocus();
            return;
        }

        abriuTelaPrincipal = false;

        botaoConectar.setEnabled(false);
        botaoUsarSemConexao.setEnabled(false);

        statusConexao.setText("Conectando ao servidor...");

        if (conexao == null) {
            conexao = new ServicoConexao(criarCallback());
        } else {
            conexao.definirCallback(criarCallback());
        }

        conexao.conectar(codigo);
    }

    private ServicoConexao.Callback criarCallback() {

        return new ServicoConexao.Callback() {

            @Override
            public void onConectado() {
                runOnUiThread(() -> {
                    if (isFinishing() || isDestroyed()) return;

                    Log.d(TAG, "Conexão WebSocket estabelecida.");

                    statusConexao.setText(
                            "Servidor conectado. Entrando na sala..."
                    );
                });
            }

            @Override
            public void onDesconectado(String motivo) {
                runOnUiThread(() -> {
                    if (isFinishing() || isDestroyed()) return;

                    if (!abriuTelaPrincipal) {
                        statusConexao.setText(
                                "Conexão encerrada. Tente novamente."
                        );
                        liberarTelaConexao();
                    }
                });
            }

            @Override
            public void onMensagem(JSONObject mensagem) {
                runOnUiThread(() -> {
                    if (isFinishing() || isDestroyed()) return;
                    processarMensagem(mensagem);
                });
            }

            @Override
            public void onErro(String erro) {
                runOnUiThread(() -> {
                    if (isFinishing() || isDestroyed()) return;

                    Log.e(TAG, erro);

                    if (!abriuTelaPrincipal) {
                        statusConexao.setText("Erro: " + erro);
                        liberarTelaConexao();
                    }
                });
            }
        };
    }

    private void processarMensagem(JSONObject mensagem) {

        String tipo = mensagem.optString("type", "");

        Log.d(TAG, "Evento recebido do servidor: " + mensagem);

        switch (tipo) {

            case "joined": {
                int usuarios = mensagem.optInt("users", 0);

                if (usuarios >= 2) {
                    abrirTelaPrincipal();
                } else {
                    statusConexao.setText(
                            "Conectado! Aguardando o segundo usuário..."
                    );
                }
                break;
            }

            case "peer_connected":
                statusConexao.setText(
                        "Parceiro conectado! Abrindo aplicativo..."
                );
                abrirTelaPrincipal();
                break;

            case "peer_disconnected":
                statusConexao.setText("Seu parceiro desconectou.");
                break;

            case "error": {
                String erro = mensagem.optString(
                        "message",
                        "O servidor informou um erro."
                );
                statusConexao.setText(erro);
                liberarTelaConexao();
                break;
            }

            default:
                Log.d(TAG, "Evento não tratado: " + tipo);
                break;
        }
    }

    private void abrirTelaPrincipal() {

        if (abriuTelaPrincipal) return;

        if (conexao == null || !conexao.estaConectado()) {
            statusConexao.setText(
                    "A conexão com o servidor foi perdida."
            );
            liberarTelaConexao();
            return;
        }

        abriuTelaPrincipal = true;

        Intent intent = new Intent(
                ConexaoActivity.this,
                MainActivity.class
        );

        intent.putExtra("codigo_par", conexao.getCodigoPar());
        intent.putExtra("papel_usuario", conexao.getPapelUsuario());
        startActivity(intent);
    }

    private void liberarTelaConexao() {

        botaoConectar.setEnabled(true);
        botaoUsarSemConexao.setEnabled(true);
        campoCodigoPar.setEnabled(true);
    }

    @Override
    protected void onResume() {

        super.onResume();
        ativarModoImersivo();

        if (!abriuTelaPrincipal && conexao != null) {
            conexao.definirCallback(criarCallback());
        }
    }

    @Override
    protected void onDestroy() {
        super.onDestroy();
        // A conexão permanece compartilhada com MainActivity.
    }
}

public class ServicoConexao {

    private static final String TAG = "ServicoConexao";

    private static final String URL_SERVIDOR =
            "wss://rpghqserver-1.onrender.com";

    public interface Callback {
        void onConectado();
        void onDesconectado(String motivo);
        void onMensagem(JSONObject mensagem);
        void onErro(String erro);
    }

    private final Handler handlerPrincipal =
            new Handler(Looper.getMainLooper());

    private volatile Callback callback;
    private volatile WebSocketClient cliente;

    private volatile boolean conectado = false;
    private volatile String codigoPar;
    private volatile String papelUsuario = "";

    private volatile boolean controleConcedido = false;
    private volatile boolean solicitacaoPendente = false;

    private final AtomicBoolean entradaEnviada =
            new AtomicBoolean(false);

    public ServicoConexao(Callback callback) {
        this.callback = callback;
    }

    public void definirCallback(Callback callback) {
        this.callback = callback;
    }

    // =====================================================
    // CONEXÃO
    // =====================================================

    public synchronized void conectar(String codigo) {

        if (codigo == null || codigo.trim().isEmpty()) {
            avisarErro("Digite o código da sala.");
            return;
        }

        if (estaConectado()) {
            Log.d(TAG, "Já existe uma conexão ativa.");
            avisarConectado();
            return;
        }

        if (cliente != null) {
            ReadyState estado = cliente.getReadyState();

            if (estado == ReadyState.OPEN
                    || estado == ReadyState.NOT_YET_CONNECTED) {
                Log.d(TAG, "Conexão já está em andamento.");
                return;
            }
        }

        codigoPar = codigo.trim();
        papelUsuario = "";

        conectado = false;
        controleConcedido = false;
        solicitacaoPendente = false;

        entradaEnviada.set(false);

        try {

            cliente = new WebSocketClient(new URI(URL_SERVIDOR)) {

                @Override
                public void onOpen(ServerHandshake handshake) {

                    conectado = true;

                    Log.d(TAG, "WebSocket conectado.");

                    enviarEntradaNaSala();
                    avisarConectado();
                }

                @Override
                public void onMessage(String mensagemRecebida) {

                    Log.d(
                            TAG,
                            "Mensagem recebida: " + mensagemRecebida
                    );

                    try {

                        JSONObject mensagem =
                                new JSONObject(mensagemRecebida);

                        processarEstadoDaConexao(mensagem);
                        avisarMensagem(mensagem);

                    } catch (JSONException e) {

                        Log.e(
                                TAG,
                                "Mensagem JSON inválida.",
                                e
                        );

                        avisarErro("Mensagem recebida inválida.");
                    }
                }

                @Override
                public void onClose(
                        int codigoFechamento,
                        String motivo,
                        boolean remoto) {

                    conectado = false;
                    controleConcedido = false;
                    solicitacaoPendente = false;

                    entradaEnviada.set(false);

                    Log.d(
                            TAG,
                            "WebSocket encerrado: " + motivo
                    );

                    avisarDesconectado(motivo);
                }

                @Override
                public void onError(Exception erro) {

                    conectado = false;
                    controleConcedido = false;
                    solicitacaoPendente = false;

                    Log.e(TAG, "Erro no WebSocket.", erro);

                    String mensagem = erro.getMessage();

                    if (mensagem == null
                            || mensagem.trim().isEmpty()) {

                        mensagem =
                                "Erro na conexão com o servidor.";
                    }

                    avisarErro(mensagem);
                }
            };

            cliente.connect();

        } catch (Exception e) {

            conectado = false;
            controleConcedido = false;
            solicitacaoPendente = false;

            Log.e(
                    TAG,
                    "Não foi possível iniciar a conexão.",
                    e
            );

            avisarErro(
                    "Não foi possível iniciar a conexão: "
                            + e.getMessage()
            );
        }
    }

    // =====================================================
    // PROCESSAMENTO DAS MENSAGENS DO SERVIDOR
    // =====================================================

    private void processarEstadoDaConexao(JSONObject mensagem) {

        String tipo = mensagem.optString("type", "");

        switch (tipo) {

            case "joined":

                papelUsuario =
                        mensagem.optString("role", "");

                Log.d(
                        TAG,
                        "Papel atribuído pelo servidor: "
                                + papelUsuario
                );

                break;

            case "control_granted":

                controleConcedido = true;
                solicitacaoPendente = false;

                Log.d(
                        TAG,
                        "Controle autorizado pelo servidor."
                );

                break;

            case "control_busy":

                controleConcedido = false;
                solicitacaoPendente = false;

                Log.d(
                        TAG,
                        "Controle ocupado pelo outro usuário."
                );

                break;

            case "control_released":

                controleConcedido = false;
                solicitacaoPendente = false;

                Log.d(
                        TAG,
                        "Controle liberado pelo servidor."
                );

                break;

            case "peer_disconnected":

                controleConcedido = false;
                solicitacaoPendente = false;

                Log.d(
                        TAG,
                        "O outro usuário desconectou."
                );

                break;

            case "control":

                if (!mensagem.optBoolean("active", false)) {

                    controleConcedido = false;
                    solicitacaoPendente = false;
                }

                break;

            case "error":

                solicitacaoPendente = false;

                Log.e(
                        TAG,
                        "Erro informado pelo servidor: "
                                + mensagem.optString("message", "")
                );

                break;
        }
    }

    // =====================================================
    // ENTRADA NA SALA
    // =====================================================

    private void enviarEntradaNaSala() {

        if (!conectado
                || codigoPar == null
                || !entradaEnviada.compareAndSet(false, true)) {

            return;
        }

        try {

            JSONObject mensagem = new JSONObject();

            mensagem.put("type", "join");
            mensagem.put("pair", codigoPar);

            enviarJson(mensagem);

            Log.d(
                    TAG,
                    "Entrada na sala enviada: " + codigoPar
            );

        } catch (JSONException e) {

            entradaEnviada.set(false);

            Log.e(
                    TAG,
                    "Erro ao preparar entrada na sala.",
                    e
            );

            avisarErro("Erro ao entrar na sala.");
        }
    }

    // =====================================================
    // ENVIAR CONTROLE
    // =====================================================

    public void enviarControle(boolean ativo, float progresso) {

        if (!estaConectado()) {

            Log.w(
                    TAG,
                    "Controle não enviado: conexão indisponível."
            );

            return;
        }

        if (ativo && !controleConcedido) {
            solicitacaoPendente = true;
        }

        if (!ativo) {
            controleConcedido = false;
            solicitacaoPendente = false;
        }

        float progressoLimitado = Math.max(
                0.0f,
                Math.min(1.0f, progresso)
        );

        try {

            JSONObject mensagem = new JSONObject();

            mensagem.put("type", "control");
            mensagem.put("active", ativo);
            mensagem.put("progress", progressoLimitado);

            enviarJson(mensagem);

            Log.d(
                    TAG,
                    "Controle enviado: " + mensagem
            );

        } catch (JSONException e) {

            Log.e(
                    TAG,
                    "Erro ao preparar comando de controle.",
                    e
            );

            avisarErro(
                    "Não foi possível preparar o comando."
            );
        }
    }

    // =====================================================
    // ENVIO JSON
    // =====================================================

    private synchronized void enviarJson(JSONObject mensagem) {

        WebSocketClient conexaoAtual = cliente;

        if (conexaoAtual == null
                || conexaoAtual.getReadyState()
                != ReadyState.OPEN) {

            Log.w(
                    TAG,
                    "Mensagem não enviada: WebSocket fechado."
            );

            return;
        }

        try {

            conexaoAtual.send(mensagem.toString());

            Log.d(
                    TAG,
                    "Mensagem enviada: " + mensagem
            );

        } catch (Exception e) {

            Log.e(
                    TAG,
                    "Erro ao enviar mensagem.",
                    e
            );

            avisarErro(
                    "Falha ao enviar dados ao parceiro."
            );
        }
    }

    // =====================================================
    // CONSULTAS DE ESTADO
    // =====================================================

    public boolean estaConectado() {

        WebSocketClient conexaoAtual = cliente;

        return conectado
                && conexaoAtual != null
                && conexaoAtual.getReadyState()
                == ReadyState.OPEN;
    }

    public String getCodigoPar() {
        return codigoPar;
    }

    public String getPapelUsuario() {
        return papelUsuario;
    }

    public boolean possuiControle() {
        return controleConcedido;
    }

    public boolean possuiSolicitacaoPendente() {
        return solicitacaoPendente;
    }

    // =====================================================
    // DESCONECTAR
    // =====================================================

    public synchronized void desconectar() {

        conectado = false;
        controleConcedido = false;
        solicitacaoPendente = false;

        entradaEnviada.set(false);

        WebSocketClient conexaoAtual = cliente;
        cliente = null;

        if (conexaoAtual != null) {

            try {

                if (conexaoAtual.isOpen()) {
                    conexaoAtual.close();
                }

            } catch (Exception e) {

                Log.e(
                        TAG,
                        "Erro ao fechar WebSocket.",
                        e
                );
            }
        }
    }

    // =====================================================
    // CALLBACKS
    // =====================================================

    private void avisarConectado() {

        handlerPrincipal.post(() -> {

            Callback atual = callback;

            if (atual != null) {
                atual.onConectado();
            }
        });
    }

    private void avisarDesconectado(String motivo) {

        handlerPrincipal.post(() -> {

            Callback atual = callback;

            if (atual != null) {
                atual.onDesconectado(motivo);
            }
        });
    }

    private void avisarMensagem(JSONObject mensagem) {

        handlerPrincipal.post(() -> {

            Callback atual = callback;

            if (atual != null) {
                atual.onMensagem(mensagem);
            }
        });
    }

    private void avisarErro(String erro) {

        handlerPrincipal.post(() -> {

            Callback atual = callback;

            if (atual != null) {
                atual.onErro(erro);
            }
        });
    }
}

package com.example.rpghqconspiration;

import android.content.Context;
import android.graphics.Canvas;
import android.graphics.Color;
import android.graphics.Movie;
import android.graphics.Paint;
import android.util.AttributeSet;

import androidx.appcompat.widget.AppCompatImageView;

import java.io.File;
import java.io.FileInputStream;

public class GifSincronizadoView extends AppCompatImageView {

    private Movie movie;

    private final Paint paint = new Paint(
            Paint.ANTI_ALIAS_FLAG |
                    Paint.FILTER_BITMAP_FLAG
    );

    private int duracaoMs = 1000;
    private int posicaoMs = 0;
    private boolean reproduzindo = false;

    public GifSincronizadoView(Context context) {
        super(context);
        inicializar();
    }

    public GifSincronizadoView(
            Context context,
            AttributeSet attrs
    ) {
        super(context, attrs);
        inicializar();
    }

    public GifSincronizadoView(
            Context context,
            AttributeSet attrs,
            int defStyleAttr
    ) {
        super(context, attrs, defStyleAttr);
        inicializar();
    }

    private void inicializar() {
        setLayerType(LAYER_TYPE_SOFTWARE, null);
        setBackgroundColor(Color.BLACK);
        setScaleType(ScaleType.FIT_CENTER);
    }

    public synchronized boolean carregarGif(File arquivo) {

        if (arquivo == null ||
                !arquivo.exists() ||
                arquivo.length() <= 0) {
            return false;
        }

        try (FileInputStream entrada =
                     new FileInputStream(arquivo)) {

            Movie novoMovie = Movie.decodeStream(entrada);

            if (novoMovie == null) {
                return false;
            }

            movie = novoMovie;

            int duracao = movie.duration();

            duracaoMs = duracao > 0 ? duracao : 1000;

            posicaoMs = 0;
            reproduzindo = false;

            postInvalidate();

            return true;

        } catch (Exception e) {

            movie = null;

            return false;
        }
    }

    public synchronized void definirPosicao(int posicao) {

        if (movie == null) return;

        posicaoMs = limitarPosicao(posicao);

        postInvalidate();
    }

    public synchronized void iniciar() {

        if (movie == null) return;

        reproduzindo = true;

        postInvalidate();
    }

    public synchronized void pausar() {

        reproduzindo = false;

        postInvalidate();
    }

    public synchronized void reiniciar() {

        posicaoMs = 0;

        reproduzindo = false;

        postInvalidate();
    }

    public synchronized int getDuracaoMs() {
        return duracaoMs;
    }

    public synchronized int getPosicaoMs() {
        return posicaoMs;
    }

    public synchronized boolean estaReproduzindo() {
        return reproduzindo;
    }

    public synchronized boolean possuiGif() {
        return movie != null;
    }

    private int limitarPosicao(int posicao) {

        return Math.max(
                0,
                Math.min(posicao, duracaoMs - 1)
        );
    }

    @Override
    protected synchronized void onDraw(Canvas canvas) {

        if (movie == null) {
            super.onDraw(canvas);
            return;
        }

        canvas.drawColor(Color.BLACK);

        int largura = getWidth();
        int altura = getHeight();

        if (largura <= 0 || altura <= 0) return;

        int larguraGif = movie.width();
        int alturaGif = movie.height();

        if (larguraGif <= 0 || alturaGif <= 0) return;

        movie.setTime(limitarPosicao(posicaoMs));

        float escala = Math.min(
                (float) largura / larguraGif,
                (float) altura / alturaGif
        );

        float x = (largura - larguraGif * escala) / 2f;
        float y = (altura - alturaGif * escala) / 2f;

        canvas.save();
        canvas.translate(x, y);
        canvas.scale(escala, escala);

        movie.draw(canvas, 0, 0, paint);

        canvas.restore();
    }
}

import android.os.Build;
import android.view.View;
import android.view.Window;
import android.view.WindowInsets;
import android.view.WindowInsetsController;

public final class ModoImersivo {

    private ModoImersivo() {
        // Impede a criação de instâncias.
    }

    public static void ativar(Window window) {

        if (window == null) {
            return;
        }

        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.R) {

            WindowInsetsController controller =
                    window.getInsetsController();

            if (controller != null) {

                controller.hide(
                        WindowInsets.Type.statusBars() |
                                WindowInsets.Type.navigationBars()
                );

                controller.setSystemBarsBehavior(
                        WindowInsetsController
                                .BEHAVIOR_SHOW_TRANSIENT_BARS_BY_SWIPE
                );
            }

        } else {

            window.getDecorView().setSystemUiVisibility(
                    View.SYSTEM_UI_FLAG_FULLSCREEN |
                            View.SYSTEM_UI_FLAG_HIDE_NAVIGATION |
                            View.SYSTEM_UI_FLAG_IMMERSIVE_STICKY |
                            View.SYSTEM_UI_FLAG_LAYOUT_FULLSCREEN |
                            View.SYSTEM_UI_FLAG_LAYOUT_HIDE_NAVIGATION |
                            View.SYSTEM_UI_FLAG_LAYOUT_STABLE
            );
        }
    }
}

package com.example.rpghqconspiration;

import android.util.Log;

import org.json.JSONObject;

public final class GerenciadorSessao {

    private static final String TAG = "GerenciadorSessao";

    private ServicoConexao servicoConexao;

    private Listener listener;

    private ServicoConexao.Callback callbackRegistrado;

    // =====================================================
    // INTERFACE DE COMUNICAÇÃO COM A MAINACTIVITY
    // =====================================================

    public interface Listener {

        boolean atividadeDisponivel();

        void onConectado();

        void onDesconectado(String motivo);

        void onMensagem(JSONObject mensagem);

        void onErro(String erro);
    }

    // =====================================================
    // CONSTRUTOR
    // =====================================================

    public GerenciadorSessao() {
    }

    // =====================================================
    // REGISTRAR CALLBACKS
    // =====================================================

    public void registrarCallbacks(
            ServicoConexao servico,
            Listener novoListener) {

        if (servico == null || novoListener == null) {

            Log.e(TAG, "Serviço ou listener inválido.");

            return;
        }

        this.servicoConexao = servico;
        this.listener = novoListener;

        callbackRegistrado = new ServicoConexao.Callback() {

                Listener atual = obterListenerAtivo();

                if (atual != null) {

                }
            }

            public void onDesconectado(String motivo) {

                Listener atual = obterListenerAtivo();

                if (atual != null) {
                    atual.onDesconectado(motivo);
                }
            }

            @Override
            public void onMensagem(JSONObject mensagem) {

                Listener atual = obterListenerAtivo();

                if (atual != null) {
                    atual.onMensagem(mensagem);
                }
            }

            @Override
            public void onErro(String erro) {

                Listener atual = obterListenerAtivo();

                if (atual != null) {
                    atual.onErro(erro);
                }
            }
        };

        servicoConexao.definirCallback(callbackRegistrado);

        Log.d(TAG, "Callbacks registrados.");
    }

    // =====================================================
    // VERIFICAR DISPONIBILIDADE DA ATIVIDADE
    // =====================================================

    private Listener obterListenerAtivo() {

        Listener atual = listener;

        if (atual == null) {
            return null;
        }

        if (!atual.atividadeDisponivel()) {
            return null;
        }

        return atual;
    }

    // =====================================================
    // REMOVER CALLBACK
    // =====================================================

    public void removerCallback() {

        if (servicoConexao != null) {

            servicoConexao.definirCallback(null);
        }

        callbackRegistrado = null;

        listener = null;

        servicoConexao = null;

        Log.d(TAG, "Callbacks removidos.");
    }

package com.example.rpghqconspiration;

import android.util.Log;

/**
 * Gerencia a contagem dos ciclos de movimentação
 * e decide o que fazer quando os controles são liberados.
 *
 * Não controla diretamente Views, MotionEvents ou GIFs.
 */
public final class GerenciadorMovimento {

    private static final String TAG = "GerenciadorMovimento";

    // =====================================================
    // CONFIGURAÇÕES
    // =====================================================

    private static final int CICLOS_NECESSARIOS = 3;

    private static final float LIMITE_EXTREMO_SUPERIOR = 0.15f;

    private static final float LIMITE_EXTREMO_INFERIOR = 0.85f;

    // =====================================================
    // ESTADO DA MOVIMENTAÇÃO
    // =====================================================

    private int ciclosConcluidos = 0;

    private int extremoInicial = -1;

    private boolean alcancouExtremoOposto = false;

    // =====================================================
    // COMUNICAÇÃO COM A MAINACTIVITY
    // =====================================================

    public interface Listener {

        boolean possuiGif();

        void avancarParaProximoGif();

        void retomarAnimacao();
    }

    private final Listener listener;

    // =====================================================
    // CONSTRUTOR
    // =====================================================

    public GerenciadorMovimento(Listener listener) {

        if (listener == null) {
            throw new IllegalArgumentException(
                    "O Listener não pode ser nulo."
            );
        }

        this.listener = listener;
    }

    // =====================================================
    // REGISTRAR PROGRESSO DO MOVIMENTO
    // =====================================================

    public void registrarProgresso(
            float progresso,
            boolean controleManualAtivo,
            boolean controleRemotoAtivo) {

        if (!controleManualAtivo &&
                !controleRemotoAtivo) {

            return;
        }

        if (Float.isNaN(progresso) ||
                Float.isInfinite(progresso)) {

            return;
        }

        progresso = Math.max(
                0f,
                Math.min(1f, progresso)
        );

        int extremoAtual = -1;

        if (progresso <= LIMITE_EXTREMO_SUPERIOR) {

            extremoAtual = 0;

        } else if (progresso >= LIMITE_EXTREMO_INFERIOR) {

            extremoAtual = 1;
        }

        if (extremoAtual == -1) {
            return;
        }

        if (extremoInicial == -1) {

            extremoInicial = extremoAtual;

            return;
        }

        if (!alcancouExtremoOposto &&
                extremoAtual != extremoInicial) {

            alcancouExtremoOposto = true;

            return;
        }

        if (alcancouExtremoOposto &&
                extremoAtual == extremoInicial) {

            ciclosConcluidos++;

            alcancouExtremoOposto = false;

            Log.d(
                    TAG,
                    "Ciclo completo: " +
                            ciclosConcluidos +
                            "/" +
                            CICLOS_NECESSARIOS
            );
        }
    }

    // =====================================================
    // REINICIAR CONTAGEM
    // =====================================================

    public void reiniciarContagem() {

        ciclosConcluidos = 0;

        extremoInicial = -1;

        alcancouExtremoOposto = false;

        Log.d(
                TAG,
                "Contagem de movimentos reiniciada."
        );
    }

    // =====================================================
    // FINALIZAR MOVIMENTO
    // =====================================================

    public void finalizarMovimentoSeLiberado(
            boolean esquerdoPressionado,
            boolean direitoPressionado,
            boolean controleManualAtivo,
            boolean controleRemotoAtivo) {

        // Ainda existe algum controle em utilização.

        if (esquerdoPressionado ||
                direitoPressionado ||
                controleManualAtivo ||
                controleRemotoAtivo) {

            return;
        }

        // Não existe GIF carregado.

        if (!listener.possuiGif()) {

            reiniciarContagem();

            return;
        }

        // Movimento suficiente para avançar.

        if (ciclosConcluidos >= CICLOS_NECESSARIOS) {

            Log.d(
                    TAG,
                    "Três ciclos concluídos. Avançando para o próximo GIF."
            );

            reiniciarContagem();

            listener.avancarParaProximoGif();

            return;
        }

        // Movimento insuficiente: mantém o GIF atual.

        Log.d(
                TAG,
                "Movimento insuficiente: " +
                        ciclosConcluidos +
                        "/" +
                        CICLOS_NECESSARIOS +
                        ". Mantendo o GIF atual."
        );

        reiniciarContagem();

        listener.retomarAnimacao();
    }

    // =====================================================
    // CONSULTA DO PROGRESSO
    // =====================================================

    public int getCiclosConcluidos() {

        return ciclosConcluidos;
    }

    public int getCiclosNecessarios() {

        return CICLOS_NECESSARIOS;
    }
}

                }
            }

            if (conexao != null) {

            }
        }
    }

package com.example.rpghqconspiration;

import java.util.ArrayList;

/**
 * Responsável por interpretar linhas no formato CSV.
 *
 * Identifica vírgulas que separam colunas, respeitando
 * vírgulas dentro de campos delimitados por aspas.
 */
public final class CsvParser {

    private CsvParser() {
        // Impede a criação de instâncias.
    }

    /**
     * Separa uma linha CSV em suas respectivas colunas.
     *
     * Também interpreta aspas duplas escapadas ("").
     */
    public static ArrayList<String> separar(String linha) {

        ArrayList<String> resultado =
                new ArrayList<>();

        StringBuilder campo =
                new StringBuilder();

        boolean dentroAspas = false;

        for (int i = 0; i < linha.length(); i++) {

            char caractere = linha.charAt(i);

            if (caractere == '"') {

                if (dentroAspas &&
                        i + 1 < linha.length() &&
                        linha.charAt(i + 1) == '"') {

                    campo.append('"');

                    i++;

                } else {

                    dentroAspas = !dentroAspas;
                }

            } else if (caractere == ',' &&
                    !dentroAspas) {

                resultado.add(campo.toString());

                campo.setLength(0);

            } else {

                campo.append(caractere);
            }
        }

        resultado.add(campo.toString());

        return resultado;
    }
}

                return null;
            }

            return resultado;

        } catch (Exception e) {

        }
    }

public final class GoogleDriveDownloader {

    private GoogleDriveDownloader() {

    }

    /**
     * Baixa um GIF e retorna o arquivo salvo.
     *
     * @param id ID do arquivo no Google Drive.
     * @param pastaGifs Pasta onde os GIFs são armazenados.
     */
    public static File baixar(
            String id,
            File pastaGifs) throws Exception {

        if (id == null || id.trim().isEmpty()) {
            throw new Exception("ID do arquivo inválido.");
        }

        if (pastaGifs == null) {
            throw new Exception("Pasta de GIFs inválida.");
        }

        if (!pastaGifs.exists() && !pastaGifs.mkdirs()) {
            throw new Exception("Não foi possível criar a pasta de GIFs.");
        }

        File arquivo = new File(
                pastaGifs,
                id + ".gif"
        );

        File temporario = new File(
                pastaGifs,
                id + ".part"
        );

        HttpURLConnection conexao = null;

        try {

            String endereco =
                    "https://drive.usercontent.google.com/download?id=" +
                            id +
                            "&export=download&confirm=t";

            conexao = (HttpURLConnection)
                    new URL(endereco).openConnection();

            conexao.setConnectTimeout(20000);
            conexao.setReadTimeout(30000);
            conexao.setInstanceFollowRedirects(true);

            int codigoHttp = conexao.getResponseCode();

            if (codigoHttp != HttpURLConnection.HTTP_OK) {
                throw new Exception("Erro HTTP: " + codigoHttp);
            }

            try (
                    InputStream entrada = conexao.getInputStream();

                    FileOutputStream saida =
                            new FileOutputStream(temporario)
            ) {

                byte[] buffer = new byte[8192];

                int quantidade;

                while ((quantidade = entrada.read(buffer)) != -1) {
                    saida.write(buffer, 0, quantidade);
                }

                saida.flush();
            }

            if (temporario.length() == 0) {
                throw new Exception("Arquivo vazio.");
            }

            try (
                    FileInputStream entradaValidacao =
                            new FileInputStream(temporario)
            ) {

                Movie teste = Movie.decodeStream(
                        entradaValidacao
                );

                if (teste == null) {
                    throw new Exception(
                            "O arquivo recebido não é um GIF válido."
                    );
                }
            }

            if (arquivo.exists() && !arquivo.delete()) {
                throw new Exception(
                        "Não foi possível substituir o arquivo antigo."
                );
            }

            if (!temporario.renameTo(arquivo)) {
                throw new Exception("Erro ao salvar GIF.");
            }

            return arquivo;

        } catch (Exception e) {

            if (temporario.exists()) {
                temporario.delete();
            }

            throw e;

        } finally {

            if (conexao != null) {
                conexao.disconnect();
            }
        }
    }
}

package com.example.rpghqconspiration;

import android.graphics.Bitmap;
import android.graphics.Canvas;
import android.graphics.Color;
import android.graphics.Movie;
import android.graphics.Paint;
import android.widget.ImageView;

/**
 * Responsável por desenhar os quadros dos GIFs.
 *
 * Controla o Bitmap e o Canvas utilizados
 * para apresentar a animação na ImageView.
 */
public class GifFrameRenderer {

    private final ImageView imagemGif;

    private Bitmap quadroBitmap;
    private Canvas quadroCanvas;

    public GifFrameRenderer(ImageView imagemGif) {
        this.imagemGif = imagemGif;
    }

    public void desenhar(
            Movie movie,
            long posicaoMs,
            Paint pincel) {

        if (movie == null ||
                imagemGif == null ||
                imagemGif.getWidth() <= 0 ||
                imagemGif.getHeight() <= 0) {

            return;
        }

        int largura = imagemGif.getWidth();
        int altura = imagemGif.getHeight();

        if (quadroBitmap == null ||
                quadroBitmap.getWidth() != largura ||
                quadroBitmap.getHeight() != altura) {

            liberarBitmap();

            quadroBitmap = Bitmap.createBitmap(
                    largura,
                    altura,
                    Bitmap.Config.ARGB_8888
            );

            quadroCanvas = new Canvas(quadroBitmap);

            imagemGif.setImageBitmap(quadroBitmap);
        }

        quadroCanvas.drawColor(Color.BLACK);

        int larguraGif = movie.width();
        int alturaGif = movie.height();

        if (larguraGif <= 0 || alturaGif <= 0) {
            return;
        }

        float escala = Math.min(
                (float) largura / larguraGif,
                (float) altura / alturaGif
        );

        float larguraFinal = larguraGif * escala;
        float alturaFinal = alturaGif * escala;

        float deslocamentoX =
                (largura - larguraFinal) / 2f;

        float deslocamentoY =
                (altura - alturaFinal) / 2f;

        quadroCanvas.save();

        quadroCanvas.translate(
                deslocamentoX,
                deslocamentoY
        );

        quadroCanvas.scale(escala, escala);

        movie.setTime((int) posicaoMs);

        movie.draw(
                quadroCanvas,
                0,
                0,
                pincel
        );

        quadroCanvas.restore();

        imagemGif.invalidate();
    }

    public void liberar() {

        liberarBitmap();

        quadroCanvas = null;
    }

    private void liberarBitmap() {

        if (quadroBitmap != null) {

            if (!quadroBitmap.isRecycled()) {
                quadroBitmap.recycle();
            }

            quadroBitmap = null;
        }
    }
}

    public GerenciadorMensagens(Listener listener) {

        if (listener == null) {

            throw new IllegalArgumentException(
                    "O Listener não pode ser nulo."
            );
        }

        this.listener = listener;
    }

    // =====================================================
    // PROCESSAR MENSAGEM
    // =====================================================

    public void processar(JSONObject dados) {

        if (dados == null) {

            Log.e(
                    TAG,
                    "Mensagem recebida é nula."
            );

            return;
        }

        String tipo = dados.optString("type");

        Log.d(
                TAG,
                "Processando mensagem: " + dados
        );

        switch (tipo) {

            case "joined": {

                int usuarios = dados.optInt("users", 0);

                listener.onJoined(usuarios);

                break;
            }

            case "peer_connected": {

                listener.onPeerConnected();

                break;
            }

            case "peer_disconnected": {

                listener.onPeerDisconnected();

                break;
            }

            case "unlock_b": {

                listener.onUnlockB();

                break;
            }

            case "error": {

                String mensagem =
                        dados.optString("message");

                listener.onError(mensagem);

                break;
            }

            case "control": {

                listener.onControl(dados);

                break;
            }

            default: {

                Log.d(
                        TAG,
                        "Mensagem não reconhecida: " + tipo
                );

                listener.onMensagemDesconhecida(tipo);

                break;
            }
        }
    }
}
